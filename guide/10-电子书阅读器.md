## 10.1 系统架构

### 10.1.1 三层架构

```
┌─────────────────────────────────────────┐
│              页面壳（状态总线）              │
│  read.tsx（选区/PopBar/划线/想法/听书）     │
├─────────────────────────────────────────┤
│              渲染器层                      │
│  EpubPane.tsx / PdfPane.tsx              │
├─────────────────────────────────────────┤
│              工具层                        │
│  mark/ reader/ listen/ pdf/              │
└─────────────────────────────────────────┘
```

### 10.1.2 M1-M8 落地阶段

| 阶段 | 目标 | 关键文件 |
|------|------|----------|
| M1 | 渲染管线 | `EpubPane.tsx` |
| M2 | 阅读设置 | `epubReaderSettings.ts` |
| M3 | 目录与导航 | `epubScrolledNav.ts`、`epubSpineIndex.ts` |
| M4 | 选区 + PopBar | `epubSelectionToolbarAttach.ts` |
| M5 | 用户划线 | `epubUserHighlights.ts` |
| M6 | 想法下划线 | `epubThoughtAnnotations.ts`、`epubThoughtCluster.ts` |
| M7 | 听书 | `useEpubChapterListen.ts`、`useEbookQuoteListen.ts` |
| M8 | PDF + 持久化 + 同步 | `pdfReaderSettings.ts`、`usePublicEbookThoughtSync.ts` |

### 10.1.3 核心设计原则

- **epub.js 是单一渲染源**
- 划线/想法/听书都是**叠加层**（annotations API + iframe 内 SVG marks-pane）
- **不改原文 DOM**，只在上面叠加
- 阅读进度防抖 800ms 落库
- Tauri 桌面端读本地文件，Web 端走云端

## 10.2 EPUB 渲染管线

### 10.2.1 EpubPane 生命周期

```typescript
// views/ebook/components/reader/EpubPane.tsx
export function EpubPane({ fileUrl, settings, onSelection, onProgress }) {
  const renditionRef = useRef<rendition.Rendition | null>(null);
  const bookRef = useRef<Book | null>(null);

  // 1. 初始化 Book
  useEffect(() => {
    const book = ePub(fileUrl, {
      flow: settings.flow,  // 'paginated' | 'scrolled'
      width: '100%',
      height: '100%',
    });
    bookRef.current = book;

    // 2. 初始化 Rendition
    const rendition = book.renderTo(containerRef.current!, {
      width: '100%',
      height: '100%',
      spread: 'auto',
    });

    // 3. 应用阅读设置
    applyReaderSettings(rendition, settings);

    // 4. 绑定事件
    rendition.on('relocated', (location) => {
      onProgress?.(location);
    });

    rendition.on('selection', (range) => {
      onSelection?.(range);
    });

    // 5. 显示
    rendition.display();
    renditionRef.current = rendition;

    return () => {
      rendition.destroy();
      book.destroy();
    };
  }, [fileUrl]);

  // 6. 应用注释（划线、想法）
  useEffect(() => {
    if (!renditionRef.current) return;
    syncEpubReadingAnnotations(renditionRef.current, annotations);
  }, [annotations, renditionRef.current]);
}
```

```typescript
// utils/epub/reader/epubReaderSettings.ts
interface ReaderSettings {
  fontSize: number;       // 字体大小 (16-32)
  lineHeight: number;     // 行高 (1.2-2.0)
  fontFamily: string;     // 字体
  background: string;     // 背景色
  flow: 'paginated' | 'scrolled';
  theme: 'light' | 'dark' | 'sepia';
}

function applyReaderSettings(rendition: rendition.Rendition, settings: ReaderSettings) {
  const css = `
    :root {
      --epub-font-size: ${settings.fontSize}px;
      --epub-line-height: ${settings.lineHeight};
      --epub-font-family: ${settings.fontFamily};
      --epub-background: ${settings.background};
    }
  `;

  rendition.themes.register('custom', {
    body: {
      'font-size': `${settings.fontSize}px`,
      'line-height': settings.lineHeight,
      'font-family': settings.fontFamily,
      'background': settings.background,
    },
  });

  rendition.themes.select('custom');
  rendition.themes.fontSize(`${settings.fontSize}px`);
}
```

```typescript
// utils/epub/reader/epubSelectionToolbarAttach.ts
function setupSelectionToolbar(rendition: rendition.Rendition, onSelection: (range: Range) => void) {
  rendition.on('mouseup', () => {
    const selection = rendition.getSelection();
    if (!selection) return;

    const range = selection.range;
    const rect = range.getBoundingClientRect();

    // 检查是否选到了足够文本（>= 3 字符）
    if (range.toString().trim().length < 3) return;

    onSelection(range);
  });
}
```

```typescript
type HighlightStyle = 'highlight' | 'underline' | 'wavy';

const STYLE_CLASS_MAP: Record<HighlightStyle, string> = {
  highlight: 'user-highlight-yellow',
  underline: 'user-underline-blue',
  wavy: 'user-wavy-red',

function applyUserHighlight(
  rendition: rendition.Rendition,
  highlight: Highlight,
) {
  // 使用 epub.js annotations API
  rendition.annotations.highlight(
    highlight.cfi,
    highlight.selectedText,
    {
      'data-highlight-id': highlight.id,
      class: STYLE_CLASS_MAP[highlight.style],
    },
    (annotation) => {
      annotation.storage = { type: 'persistent' };
    },
  );
}
```

## 10.5 想法下划线

### 10.5.1 懒加载与视口

```typescript
// utils/epub/mark/epubThoughtAnnotations.ts
function applyThoughtUnderlines(
  rendition: rendition.Rendition,
  thoughts: Thought[],
  viewportRange?: Range,  // 可选：仅加载视口内
) {
  // 懒加载：仅加载视口范围内的想法
  const visibleThoughts = viewportRange
    ? filterThoughtsByViewport(thoughts, viewportRange)
    : thoughts;

  for (const thought of visibleThoughts) {
    rendition.annotations.underline(
      thought.cfi,
      thought.selectedText,
      {
        'data-thought-id': thought.id,
        class: thought.pinned ? 'thought-pinned' : 'thought-underline',
      },
    );
  }
}

function filterThoughtsByViewport(thoughts: Thought[], viewportRange: Range): Thought[] {
  return thoughts.filter(thought => {
    const thoughtRange = cfiToRange(thought.cfi);
    return viewportRange.intersectsNode(thoughtRange.commonAncestorContainer);
  });
}
```

### 10.5.2 想法簇聚合

```typescript
// utils/epub/mark/epubThoughtCluster.ts
function clusterThoughts(thoughts: Thought[]): ThoughtCluster[] {
  // 按 CFI 位置排序
  const sorted = [...thoughts].sort((a, b) =>
    a.cfi.localeCompare(b.cfi),
  );

  // 贪心聚类：相邻 < 200 字符视为一簇
  const clusters: ThoughtCluster[] = [];
  let currentCluster: Thought[] = [];

  for (const thought of sorted) {
    if (currentCluster.length === 0) {
      currentCluster.push(thought);
    } else {
      const lastThought = currentCluster[currentCluster.length - 1];
      const distance = calculateDistance(lastThought.cfi, thought.cfi);
      if (distance < 200) {
        currentCluster.push(thought);
      } else {
        clusters.push({ thoughts: currentCluster });
        currentCluster = [thought];
      }
    }
  }

  if (currentCluster.length > 0) {
    clusters.push({ thoughts: currentCluster });
  }

  return clusters;
}
```

## 10.6 听书系统

### 10.6.1 听书数据流

```
用户点耳机图标
    │
    ├─ extractVisibleListenSection  → 提取当前视口段落
    │
    ├─ buildSentenceOffsetSpans     → 构建句子偏移映射
    │
    ├─ indexChapterSentenceRanges   → 章节句子索引
    │
    └─ runListenLoop                → 进入听书循环
        │
        ├─ scroll 模式：runScrollSectionLoop
        │   ├─ playSentencesFromCursor   → 逐句 TTS
        │   ├─ updateSentenceHighlight   → 淡黄句背景
        │   ├─ autoFollowScroll          → 自动跟随滚动
        │   └─ advanceScrollListenSection → 下一 iframe
        │
        └─ paginated 模式：runPaginatedListenLoop
            ├─ playSentencesFromCursor
            ├─ updateSentenceHighlight
            ├─ autoAdvancePage
            └─ advancePaginatedListenSection
```

```typescript
// utils/epub/listen/epubListenChapter.ts
async function playSentences(
  sentences: Sentence[],
  config: TtsConfig,
): Promise<void> {
  for (const sentence of sentences) {
    // 标记当前句
    highlightSentence(sentence.range, 'active');

    try {
      if (config.source === 'cloud') {
        await playCloudTts(sentence.text, config);
      } else {
        await playLocalTts(sentence.text, config);
      }
    } finally {
      // 清除高亮
      clearSentenceHighlight(sentence.range);
    }
  }
}
```

### 10.6.3 互斥控制

```typescript
// utils/epub/listen/epubListenExclusive.ts
class ListenExclusiveController {
  private activeSource: 'chapter' | 'quote' | null = null;
  private cancelFn: (() => void) | null = null;

  async startChapterListen(onCancel: () => void) {
    // 互斥：先停掉引用听读
    if (this.activeSource === 'quote') {
      this.cancelFn?.();
    }
    this.activeSource = 'chapter';
    this.cancelFn = onCancel;
  }

  async startQuoteListen(onCancel: () => void) {
    if (this.activeSource === 'chapter') {
      this.cancelFn?.();
    }
    this.activeSource = 'quote';
    this.cancelFn = onCancel;
  }

  stop() {
    this.cancelFn?.();
    this.activeSource = null;
    this.cancelFn = null;
  }
}
```

```typescript
function extractPdfOutline(pdf: PDFDocumentProxy): EbookTocItem[] {
  const outline = pdf.getOutline();
  if (!outline) return [];

  return outline.map((item: any) => ({
    id: crypto.randomUUID(),
    title: item.title,
    cfi: null,  // PDF 用 page number
    page: item.page,
    children: item.items ? extractPdfOutline({ items: item.items }) : [],
```
