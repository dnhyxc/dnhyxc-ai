# 二、插件系统

## 2.1 系统架构

### 2.1.1 分层模型

本系统采用 **Module Federation (MF)** 为核心的动态微前端架构，通过分层设计实现核心逻辑与宿主逻辑的解耦：

```
┌─────────────────────────────────────────┐
│           宿主应用 (Host)                 │
│  ┌───────────────────────────────────┐  │
│  │  PluginHostPage (容器 + 样式隔离)  │  │
│  └───────────────────────────────────┘  │
│  ┌───────────────────────────────────┐  │
│  │  PluginManager (加载/校验/生命周期) │  │
│  └───────────────────────────────────┘  │
│  ┌───────────────────────────────────┐  │
│  │  HostBridge (通信/导航/数据)      │  │
│  └───────────────────────────────────┘  │
│  ┌───────────────────────────────────┐  │
│  │  Module Federation Runtime        │  │
│  └───────────────────────────────────┘  │
├─────────────────────────────────────────┤
│           插件 (Remote)                   │
│  ┌───────────────────────────────────┐  │
│  │  PluginDescriptor (元数据)         │  │
│  └───────────────────────────────────┘  │
│  ┌───────────────────────────────────┐  │
│  │  expose: ./App (组件/逻辑)        │  │
│  └───────────────────────────────────┘  │
│  ┌───────────────────────────────────┐  │
│  │  activate/deactivate (生命周期)   │  │
│  └───────────────────────────────────┘  │
└─────────────────────────────────────────┘
```

### 2.1.2 核心模块

| 模块 | 路径 | 职责 |
|------|------|------|
| **PluginManager** | `src/plugins/core/PluginManager.ts` | 插件加载、校验、生命周期管理 |
| **PluginVerifier** | `src/plugins/core/PluginVerifier.ts` | 插件合法性验证 |
| **HostBridge** | `src/plugins/bridge/HostBridge.ts` | 宿主与插件间通信 |
| **PluginHostPage** | `src/plugins/host/PluginHostPage.tsx` | 插件容器、样式隔离 |
| **styleIsolation** | `src/plugins/host/styleIsolation.ts` | CSS 样式隔离核心 |
| **federation-kit** | `packages/federation-kit/` | 共享逻辑包 |

### 2.1.3 信任等级

| 等级 | 技术方案 | 隔离程度 | 适用场景 |
|------|----------|---------|---------|
| **trusted** | MF + CSS @scope | 样式级 | 可信内部插件 |
| **untrusted** | iframe 完全隔离 | 完全 | 不可信第三方插件 |

### 2.2.1 宿主配置

```typescript

  },

```

### 2.2.2 插件配置


### 2.2.3 动态加载流程

```
1. PluginManager.ensurePlugin(pluginId)
   │
   ├─ verifyPlugin(meta)          // 校验元数据
   ├─ registerRemote(meta)       // MF Runtime 注册
   │   └─ __federation_method_setRemote({ entry, name })
   │
   ├─ beginPluginStyleCapture()   // 开启样式捕获
   │
   ├─ loadRemoteApp(meta)         // MF 动态加载
   │   └─ __federation_method_remotes[url].get(module)
   │       → Promise<{ default: Component, activate?, deactivate? }>
   │
   ├─ endCapture()                // 结束样式捕获
   ├─ mod.activate?.(bridge.api)  // 调用激活钩子
   │
   └─ 状态 → 'activated'
```

### 2.2.4 PluginRegistry 数据结构

```typescript
interface PluginDescriptor {
  id: string;                // 唯一标识
  name: string;              // 显示名称
  version: string;           // 版本号
  entry: string;             // remoteEntry.js URL
  expose: string;            // 暴露模块路径，如 './App'
  icon?: string;             // 图标 URL/SVG
  description?: string;      // 描述
  trust: 'trusted' | 'untrusted';  // 信任等级
  iframeUrl?: string;        // untrusted 模式 iframe URL
  menu?: {                   // 菜单配置
    label: string;
    order?: number;
  };
}
```

## 2.4 宿主接入方式

### 2.4.1 自动注入（路由 + 菜单）

```typescript
// plugins/core/PluginManager.ts
private async runLoad(meta: PluginDescriptor) {
  // 加载插件
  const mod = await loadRemoteApp(meta);
  const bridge = createHostBridge(meta, nav);

  // 自动注册路由
  if (meta.expose === './App') {
    RouteInjector.register({
      path: `/plugins/${meta.id}`,
      element: <PluginHostPage pluginId={meta.id} />,
    });
  }

  // 自动注入菜单
  if (meta.menu) {
    SidebarInjector.addItem({
      id: meta.id,
      label: meta.menu.label,
      to: `/plugins/${meta.id}`,
      icon: meta.icon,
      order: meta.menu.order,
    });
  }

  // 调用激活钩子
  await mod.activate?.(bridge.api);
}
```

### 2.4.2 手动挂载（嵌入式插件）

```typescript
// 在业务组件中手动挂载
import { usePluginHost } from '@/plugins/host/usePluginHost';

function EmbeddedPluginPanel({ pluginId }: { pluginId: string }) {
  const { loaded, bridge, status } = usePluginHost(pluginId);

  if (status === 'loading') return <div>Loading...</div>;
  if (status === 'error') return <div>Error: {loaded?.error}</div>;

  const Comp = loaded?.mod.default;
  if (!Comp) return null;

  return (
    <div data-mf-plugin={pluginId}>
      <Comp {...bridge} />
    </div>
  );
}
```

### 2.4.3 iframe 隔离（untrusted 模式）

```typescript
// PluginHostPage.tsx
if (loaded.meta.trust === 'untrusted') {
  return (
    <PluginErrorBoundary pluginId={pluginId}>
      <UntrustedIframe
        pluginId={pluginId}
        src={loaded.meta.iframeUrl}
        bridge={loaded.bridge}
      />
    </PluginErrorBoundary>
  );
}

// UntrustedIframe.tsx
function UntrustedIframe({ src, bridge }) {
  const iframeRef = useRef<HTMLIFrameElement>(null);

  useEffect(() => {
    const handler = (e: MessageEvent) => {
      if (e.data?.type === 'plugin-init') {
        iframeRef.current?.contentWindow?.postMessage({
          type: 'host-ready',
          api: bridge.api,
        }, '*');
      }
    };
    window.addEventListener('message', handler);
    return () => window.removeEventListener('message', handler);
  }, [bridge]);

  return (
    <iframe
      ref={iframeRef}
      src={src}
      className="h-full w-full border-0"
      sandbox="allow-scripts allow-same-origin"
    />
  );
}
```

## 2.5 缓存破坏与 HMR

### 2.5.1 Manifest Hash 缓存破坏

```typescript
// plugins/host/pluginEntryCacheBust.ts
export async function resolvePluginBust(meta: PluginDescriptor) {
  const { buildId } = await fetchManifestMeta(meta.entry);
  return {
    entry: meta.entry,
    cacheBustedUrl: `${meta.version}@${buildId}/remoteEntry.js`,
  };
}

async function fetchManifestMeta(entryUrl: string) {
  const manifestUrl = entryUrl.replace('remoteEntry.js', 'manifest.json');
  const res = await fetch(manifestUrl);
  if (!res.ok) return { buildId: Date.now().toString() };
  return res.json();
}
```

## 2.6 Vue 插件桥接

### 2.6.1 插件端（Vue）

```typescript
// 插件入口必须导出 mount/unmount
export async function mount(container: HTMLElement, props: any) {
  const { createApp } = await import('vue');
  const App = await import('./App.vue');
  const app = createApp(App.default || App);
  app.mount(container);
  (container as any).__vue_app__ = app;
}

export async function unmount(container: HTMLElement) {
  const app = (container as any).__vue_app__;
  if (app) {
    app.unmount();
    delete (container as any).__vue_app__;
  }
}
```

### 2.6.2 宿主端桥接

```typescript
// plugins/host/createVueHostBridge.tsx
export function createVueHostBridge(
  expose: { mount: Function; unmount: Function },
  pluginId: string,
) {
  return function VueHostBridge(props: any) {
    const elRef = useRef<HTMLDivElement>(null);

    useEffect(() => {
      if (!elRef.current) return;
      expose.mount(elRef.current, props);
      return () => expose.unmount();
    }, []);

    return <div ref={elRef} data-plugin-root />;
  };
}
```

## 2.7 插件图标系统

### 2.7.1 存储与渲染

```typescript
// components/PluginIcon.tsx
interface PluginIconProps {
  src: string;         // SVG URL 或 markup
  size?: number;
  animated?: 'spin' | 'pulse';
}

export function PluginIcon({ src, size = 24, animated }: PluginIconProps) {
  const [svgHtml, setSvgHtml] = useState<string | null>(null);
  const [error, setError] = useState(false);

  useEffect(() => {
    if (src.startsWith('<svg')) {
      // 内联 SVG
      setSvgHtml(normalizeSvgForHostIcon(src));
    } else {
      // fetch SVG
      fetch(src)
        .then(res => res.text())
        .then(text => setSvgHtml(normalizeSvgForHostIcon(text)))
        .catch(() => setError(true));
    }
  }, [src]);

  if (error || !svgHtml) {
    return <DefaultPluginIcon size={size} />;
  }

  return (
    <span
      className={animated ? `animate-${animated}` : ''}
      style={{ width: size, height: size }}
      dangerouslySetInnerHTML={{ __html: svgHtml }}
    />
  );
}
```

### 2.7.2 SVG 规范化

```typescript
function normalizeSvgForHostIcon(svgText: string): string {
  // 处理 currentColor
  let html = svgText
    .replace(/currentColor/g, 'var(--brand-accent, currentColor)');

  // 处理 stroke/fill 属性
  html = html.replace(
    /<svg([^>]*)>/,
    '<svg$1 width="100%" height="100%" fill="currentColor" stroke="currentColor">'
  );

  return html;
}
```

## 2.8 插件偏好持久化

### 2.8.1 状态流转

```
用户开启插件
    │
    ├─ POST /api/plugins/enable  → 后端存储用户偏好
    │   └─ PluginUserPrefs 表：{ user_id, enabled_ids: JSON[] }
    │
    ├─ PluginManager.ensurePlugin(pluginId)
    │   └─ loadRemoteApp → mount → status: 'activated'
    │
    ├─ RouteInjector / SidebarInjector 注册入口
    │
    └─ 前端 localStorage 缓存（首屏同步）

用户关闭插件
    │
    ├─ DELETE /api/plugins/disable → 更新后端偏好
    ├─ PluginManager.unloadPlugin(pluginId)
    │   └─ mod.deactivate?.() → unmount → status: 'unloaded'
    └─ 路由/菜单入口移除
```

### 2.8.2 失败状态处理

```typescript
// PluginHostPage.tsx
if (loaded?.status === 'failed') {
```
