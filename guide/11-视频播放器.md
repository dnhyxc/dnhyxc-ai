## 11.2 组件化重构

### 11.2.1 组件结构

```typescript
// components/VideoPlayer/
// ├── index.tsx          # 播放器主组件
// ├── ControlBar.tsx     # 控制条
// ├── ProgressBar.tsx    # 进度条
// ├── ThumbnailPreview   # 缩略图预览
// ├── PiPEntry.tsx       # 画中画入口
// ├── VolumeControl.tsx  # 音量控制
// ├── FullscreenButton.tsx # 全屏按钮
// └── plugins/           # xgplayer 自定义插件
//     ├── cinema.ts      # 影院模式
//     └── remote-demo.ts # 远程演示

// VideoPlayer.tsx
export function VideoPlayer({ src, poster, onPlay, onEnded }: Props) {
  const containerRef = useRef<HTMLDivElement>(null);
  const playerRef = useRef<Player | null>(null);
  const [isPlaying, setIsPlaying] = useState(false);
  const [currentTime, setCurrentTime] = useState(0);
  const [duration, setDuration] = useState(0);

  // 初始化播放器
  useEffect(() => {
    if (!containerRef.current) return;

    const player = new Player({
      el: containerRef.current,
      url: src,
      poster,
      playsinline: true,
      controls: false,  // 自定义控制条
    });

    // 注册自定义插件
    player.use(CinemaPlugin);
    player.use(ThumbnailPlugin);

    // 事件监听
    player.on(Player.EventTypes.PLAY, () => setIsPlaying(true));
    player.on(Player.EventTypes.PAUSE, () => setIsPlaying(false));
    player.on(Player.EventTypes.TIME_UPDATE, (e) => {
      setCurrentTime(e.currentTime);
    });
    player.on(Player.EventTypes.DURATION_CHANGE, (e) => {
      setDuration(e.duration);
    });
    player.on(Player.EventTypes.ENDED, () => onEnded?.());

    playerRef.current = player;

    return () => {
      player.destroy();
    };
  }, [src]);

  const handlePlay = () => {
    if (isPlaying) playerRef.current?.pause();
    else {
      playerRef.current?.play();
      onPlay?.();
    }
  };

  return (
    <div ref={containerRef} className="relative h-full w-full">
      <ControlBar
        isPlaying={isPlaying}
        currentTime={currentTime}
        duration={duration}
        onPlay={handlePlay}
        onSeek={(time) => playerRef.current?.seek(time)}
      />
    </div>
  );
}
```

## 11.3 影院模式

### 11.3.1 深色背景控制条

```typescript
// plugins/cinema.ts
class CinemaPlugin extends Plugin {
  static get pluginName() { return 'cinema'; }

  private cinemaMode = false;

  async render(root: HTMLElement) {
    const { root: playerRoot } = this.context;

    const toggleCinema = () => {
      this.cinemaMode = !this.cinemaMode;

      if (this.cinemaMode) {
        // 深色背景
        playerRoot.classList.add('cinema-mode');
        // 控制条文字/图标变白
        const style = document.createElement('style');
        style.textContent = `
          .cinema-mode .control-bar { color: white; }
          .cinema-mode .control-bar svg { stroke: white; fill: white; }
          .cinema-mode .progress-bar { background: rgba(255,255,255,0.3); }
          .cinema-mode .progress-filled { background: #ff6b35; }
        `;
        document.head.appendChild(style);
      } else {
        playerRoot.classList.remove('cinema-mode');
      }
    };

    // 注册到上下文
    this.context.toggleCinema = toggleCinema;
  }
}
```

## 11.4 拖拽进度

### 11.4.1 像素精度计算

```typescript
// utils/video/drag.ts
export function handleSeekByPixel(
  pixelX: number,
  barWidth: number,
  duration: number,
): number {
  // 像素宽度而非百分比
  const ratio = Math.max(0, Math.min(1, pixelX / barWidth));
  return ratio * duration;
}

export function handleDragScrub(
  player: Player,
  clientX: number,
  barRect: DOMRect,
  duration: number,
) {
  const x = clientX - barRect.left;
  const seekTime = handleSeekByPixel(x, barRect.width, duration);

  // scrubHoldUntil：抑制立即 currentTime 更新
  // 等 seek 完成后再更新
  player.seek(seekTime);
}
```

## 11.5 缩略图预览

### 11.5.1 离屏渲染

```typescript
// components/VideoPlayer/ThumbnailPreview.tsx
export function ThumbnailPreview({ videoUrl, seekTime }: Props) {
  const offscreenRef = useRef<HTMLVideoElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const [thumbnail, setThumbnail] = useState<string | null>(null);

  // 初始化离屏 video
  useEffect(() => {
    const video = offscreenRef.current;
    if (!video) return;

    video.preload = 'metadata';
    video.src = videoUrl;
  }, [videoUrl]);

  // 按时间帧捕获
  useEffect(() => {
    const video = offscreenRef.current;
    const canvas = canvasRef.current;
    if (!video || !canvas) return;

    const seekToTime = async () => {
      await new Promise<void>((resolve) => {
        const onSeeked = () => {
          video.removeEventListener('seeked', onSeeked);
          resolve();
        };
        video.addEventListener('seeked', onSeeked);
        video.currentTime = seekTime;
      });

      // 捕获帧
      const ctx = canvas.getContext('2d');
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      ctx.drawImage(video, 0, 0);

      setThumbnail(canvas.toDataURL('image/jpeg', 0.7));
    };

    seekToTime();
  }, [seekTime]);

  return thumbnail ? (
    <img
      src={thumbnail}
      alt="preview"
      className="rounded border border-white/20"
      style={{ width: 160, height: 90 }}
    />
  ) : null;
}
```

## 11.6 画中画

### 11.6.1 三级回退链

```typescript
// components/VideoPlayer/PiPEntry.tsx
async function requestPictureInPicture(video: HTMLVideoElement) {
  // 1. xgplayer 插件
  if (player.usePictureInPicture) {
    return player.usePictureInPicture();
  }

  // 2. 原生 API
  if (document.pictureInPictureEnabled) {
    try {
      return await video.requestPictureInPicture();
    } catch { /* 降级 */ }
  }

  // 3. WebKit API
  if ((video as any).webkitEnterPictureInPicture) {
    return (video as any).webkitEnterPictureInPicture();
  }

  throw new Error('Picture-in-Picture not supported');
}
```

## 11.7 多端适配

### 11.7.1 特性检测

```typescript
// utils/video/platform.ts
export function isDesktop(): boolean {
  return isTauriRuntime();
}

export function supportsFeature(feature: string): boolean {
  switch (feature) {
    case 'picture-in-picture':
      return !isDesktop() && document.pictureInPictureEnabled;
    case 'global-shortcuts':
      return isDesktop();
    case 'file-dialog':
      return isDesktop();
    case 'http-retry-all-methods':
      return isDesktop();
    default:
      return false;
  }
}
```
