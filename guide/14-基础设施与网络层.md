```typescript
// utils/http.ts
const httpClient = axios.create({
  baseURL: '/api',
  timeout: 30000,
  withCredentials: true,
  headers: {
    'Content-Type': 'application/json',
  },
});

// 请求拦截器
httpClient.interceptors.request.use((config) => {
  const token = localStorage.getItem('token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }

  // 请求计时（Tauri 桌面端支持全方法重试）
  if (isTauriRuntime()) {
    config.metadata = { startTime: Date.now() };
  }

  return config;
});

// 响应拦截器
httpClient.interceptors.response.use(
  (response) => {
    // Tauri 桌面端失败重试
    if (isTauriRuntime() && response.status >= 500) {
      const retryCount = (response.config.metadata?.retryCount || 0) + 1;
      if (retryCount < 3) {
        response.config.metadata.retryCount = retryCount;
        return new Promise(resolve => {
          setTimeout(() => resolve(httpClient.request(response.config)), retryCount * 1000);
        });
      }
    }

    return response.data;
  },
  (error) => {
    // 401 → 跳转登录
    if (error.response?.status === 401) {
      localStorage.removeItem('token');
      window.location.href = '/login';
    }

    // SSE 错误特殊处理
    if (error.config?.responseType === 'stream') {
      return handleSseError(error);
    }

    return Promise.reject(error);
  },
);
```

### 14.2.2 SSE 客户端

```typescript
// utils/sse.ts
export async function* streamFetch<T = any>(
  url: string,
  options?: RequestInit & {
    onError?: (error: Error) => void;
    onComplete?: () => void;
  },
): AsyncGenerator<SseEvent<T>> {
  const controller = new AbortController();
  const { onError, onComplete, ...fetchOptions } = options || {};

  try {
    const response = await fetch(url, {
      ...fetchOptions,
      headers: {
        'Accept': 'text/event-stream',
        'Cache-Control': 'no-cache',
        ...fetchOptions?.headers,
      },
      signal: controller.signal,
    });

    if (!response.ok) {
      throw new Error(`SSE request failed: ${response.status}`);
    }

    const reader = response.body!.getReader();
    const decoder = new TextDecoder();
    let buffer = '';

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      buffer += decoder.decode(value, { stream: true });
      const lines = buffer.split('\n');
      buffer = lines.pop() || '';

      for (const line of lines) {
        if (line.startsWith('data:')) {
          const data = line.slice(5).trim();
          if (data === '[DONE]') {
            onComplete?.();
            return;
          }

          try {
            const parsed = JSON.parse(data) as SseEvent<T>;
            yield parsed;
          } catch {
            yield { type: 'token', data } as SseEvent<T>;
          }
        }
      }
    }
  } catch (error) {
    if (error instanceof AbortError) {
      // 用户主动取消
      return;
    }
    onError?.(error as Error);
    throw error;
  }
}
```

```typescript
export async function readEpubAsBlob(path: string): Promise<Blob> {
  // Tauri 内置 read_ebook_file 命令
  const base64 = await invoke<string>('read_ebook_file', { path });
  const binary = atob(base64);
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i++) {
    bytes[i] = binary.charCodeAt(i);
  }
  return new Blob([bytes], { type: 'application/epub+zip' });
}
```

```typescript
  return new Promise((resolve) => {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = options.accept || '';
    input.multiple = options.multiple ?? true;

    input.onchange = (e) => {
      const files = Array.from((e.target as HTMLInputElement).files || []);
      resolve(files);
    };

    input.click();
  });
}

  const blob = new Blob([content]);
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = fileName;
  a.click();
  URL.revokeObjectURL(url);
}
```

```typescript
export function isTauriRuntime(): boolean {
  return typeof window !== 'undefined' &&
    '__TAURI_INTERNALS__' in window;
}

export function getRuntime(): Runtime {
  if (isTauriRuntime()) return 'tauri';
  if (typeof window !== 'undefined' && 'Capacitor' in window) return 'capacitor';
  return 'web';
}
```

```typescript
export function supportsFeature(feature: FeatureFlag): boolean {
  const runtime = getRuntime();

  switch (feature) {

    case 'global-shortcuts':
      return runtime === 'tauri';
```

### 14.7.2 错误处理

```typescript
export class AppError extends Error {
  constructor(
    public readonly code: ErrorCode,
    message: string,
    public readonly details?: Record<string, any>,
  ) {
    super(message);
    this.name = 'AppError';
  }
}

export function handleError(error: unknown): AppError {
  if (error instanceof AppError) return error;

  if (error instanceof axios.AxiosError) {
    if (error.response?.status === 401) {
      return new AppError(ErrorCodes.UNAUTHORIZED, '未授权', error.response.data);
    }
    if (error.response?.status === 429) {
      return new AppError(ErrorCodes.RATE_LIMITED, '请求过于频繁', error.response.data);
    }
    if (error.code === 'ECONNABORTED') {
      return new AppError(ErrorCodes.TIMEOUT, '请求超时');
    }
    return new AppError(ErrorCodes.NETWORK, `网络错误: ${error.message}`);
  }

  if (error instanceof DOMException && error.name === 'AbortError') {
    return new AppError(ErrorCodes.CANCELLED, '请求已取消');
  }

  return new AppError(ErrorCodes.UNKNOWN, '未知错误');
}
```

### 14.8.2 MobX Store

```typescript
// stores/chat.ts (MobX)
class ChatStore {
  @observable currentSessionId: string | null = null;
  @observable sessions: Session[] = [];
  @observable messages: Map<string, Message[]> = new Map();
  @observable isStreaming = false;
  @observable isLoading = false;

  @action
  async loadSessions() {
    this.isLoading = true;
    try {
      const sessions = await api.get('/chat/sessions');
      this.sessions = sessions;
    } finally {
      this.isLoading = false;
    }
  }

  @action
  async sendMessage(content: string) {
    if (!this.currentSessionId) {
      await this.createSession();
    }

    const sessionId = this.currentSessionId!;
    this.isStreaming = true;

    // SSE 流式
    for await (const event of streamFetch('/chat/stream', {
      method: 'POST',
      body: JSON.stringify({ sessionId, content }),
    })) {
      if (event.type === 'token') {
        const messages = this.messages.get(sessionId) || [];
        const last = messages[messages.length - 1];
        if (last?.role === 'assistant') {
          last.content += event.data;
        } else {
          messages.push({ role: 'assistant', content: event.data });
        }
        this.messages.set(sessionId, messages);
      } else if (event.type === 'done') {
        this.isStreaming = false;
      }
    }
  }

  @action
  stopStreaming() {
    this.isStreaming = false;
  }
}

export const chatStore = new ChatStore();
```

## 14.10 独立文档站（remote-docs）路由迁移

> 完整实现说明见 [docs/app/独立文档站迁移.md](../docs/app/独立文档站迁移.md)。

### 14.10.1 改动动机

「更新信息」「产品指南」「插件开发手册」三个独立页原本是主站内置 SPA 路由（`/update-info`、`/project-guide`、`/plugin-dev-guide`），正文数据写在前端 TypeScript 文件中，导致：

1. 文档维护与应用发版强耦合，改文案需重新构建前端
2. 纯文档数据（`updateInfoSections.ts` 2600+ 行等）被打进主包，拖慢首屏
3. 无法做到「只更说明、不动应用」

### 14.10.2 方案：正文外置到 remote-docs 独立站

将三个页面的正文与渲染迁移到独立仓 `micro-apps/remote-docs`，部署到专属 origin（dev `http://127.0.0.1:9013`，prod `https://dnhyxc.cn:9017`）。主站只保留入口跳转能力：

| 维度 | 改动前 | 改动后 |
|------|--------|--------|
| 路由 | `routes.ts` 内置三条 SPA 路由 | 三条路由全部移除 |
| 视图 | `views/updateInfo/`、`views/projectGuide/`、`views/pluginDevGuide/` | 三个目录整体删除 |
| 跳转 | `navigate('/xxx')` SPA 内跳转 | `openExternalUrl(getXxxAbsoluteUrl(locale))` 系统浏览器 / 新标签页 |
| 正文 | 前端 TS 数据文件 | `micro-apps/remote-docs` 独立仓 |

### 14.10.3 核心工具 `remote-docs-url.ts`

```typescript
// origin 可由 VITE_REMOTE_DOCS_ORIGIN 覆盖，dev/prod 各有缺省
export function getRemoteDocsOrigin(): string {
  const fromEnv = String(import.meta.env.VITE_REMOTE_DOCS_ORIGIN ?? '').trim().replace(/\/$/, '');
  if (fromEnv) return fromEnv;
  if (import.meta.env.DEV) return 'http://127.0.0.1:9013';
  return 'https://dnhyxc.cn:9017';
}

// 拼绝对 URL，自动追加 lang / theme / accent
export function getRemoteDocsAbsoluteUrl(
  path: string,
  locale?: Locale,
  themeName?: ThemeName | null,
  accentId?: AccentId | null,
): string {
  const origin = getRemoteDocsOrigin();
  const normalized = path.startsWith('/') ? path : `/${path}`;
  let url = origin ? `${origin}${normalized}` : normalized;
  if (locale) url = withAppLangInSearch(url, locale);
  const theme = themeName === undefined ? readThemeBootstrapSync() : themeName;
  if (theme) url = appendShareThemeQuery(url, theme);
  const accent = accentId === undefined ? readAccentBootstrapSync() : accentId;
  if (accent) url = appendShareAccentQuery(url, accent);
  return url;
}
```

三个便捷函数 `getUpdateInfoAbsoluteUrl`、`getProjectGuideAbsoluteUrl`、`getPluginDevGuideAbsoluteUrl` 分别传入对应路径常量。

### 14.10.6 回归要点

- 首页 / 下载页 / 关于页 / 插件错误卡片的三个文档入口均能打开独立站
- URL 携带 `lang` / `theme` / `accent`，独立站语言与主题与主站一致
- Tauri 在系统浏览器打开，Web 在新标签页打开
- 旧路径 `/update-info` 等不再被主站路由匹配（404 兜底）
