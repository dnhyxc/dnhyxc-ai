# 六、大模型接入

## 6.1 系统架构

### 6.1.1 createLlm 工厂函数

通过统一工厂函数 `createLlm(config, options)` 封装 LangChain `ChatOpenAI` 的创建逻辑，支持按 **preset** 隔离凭证回退链与模型名解析。

### 6.1.2 Preset 与调用方映射

| Preset | 调用方 | API Key 回退链 | 模型名回退链 |
|--------|--------|---------------|-------------|
| `chat` | `ChatService` | 硅基 → DeepSeek | SILICONFLOW_MODEL_NAME → DEEPSEEK_MODEL_NAME → 默认 GLM-5.1 |
| `assistant` | `AssistantService` | 硅基 → DeepSeek → DashScope | 同上 |
| `knowledgeQa` | `KnowledgeQaService` | 硅基 → DashScope → Qwen | 同上 |
| `englishLearning` | `EnglishLearningService` | 硅基 → DeepSeek → DashScope | 同上 |
| `ocr` | `OcrService` | 智谱专用 | OCR 专用模型 |

### 6.1.3 优先级

```
用户 DB 快照（/settings/llm 保存）
    ↓
有效会员 env（SILICONFLOW_API_KEY + SILICONFLOW_MODEL_NAME）
    ↓
非会员 env（ZHIPU_API_KEY + ZHIPU_MODEL_NAME）
    ↓
代码内默认值
```

## 6.2 核心实现

### 6.2.1 createLlm 工厂

```typescript
// utils/create-llm.ts
import { ChatOpenAI } from '@langchain/openai';

type Preset = 'chat' | 'assistant' | 'knowledgeQa' | 'englishLearning' | 'ocr';

interface CreateLlmOptions {
  maxTokensPolicy?: 'optional' | 'default';
  defaultMaxTokens?: number;
  streaming?: boolean;
  temperature?: number;
}

export function createLlm(
  config: ConfigService,
  preset: Preset,
  userId?: string,
  options: CreateLlmOptions = {},
) {
  const { streaming = true, temperature = 0.7 } = options;

  // 解析凭证
  const { apiKey, baseUrl, modelName } = resolveCredentials(config, preset, userId);

  // maxTokens 策略
  let maxTokens: number | undefined;
  if (options.maxTokensPolicy === 'optional' && options.defaultMaxTokens) {
    maxTokens = options.defaultMaxTokens;
  } else if (options.maxTokensPolicy === 'default') {
    maxTokens = options.defaultMaxTokens || 4096;
  }

  return new ChatOpenAI({
    apiKey,
    modelName,
    streaming,
    temperature,
    maxTokens,
    configuration: { baseURL },
  });
}
```

### 6.2.2 凭证解析

```typescript
function resolveCredentials(
  config: ConfigService,
  preset: Preset,
  userId?: string,
): { apiKey: string; baseUrl: string; modelName: string } {
  // 1. 优先读用户 DB 快照
  if (userId) {
    const userConfig = getUserLlmConfig(userId, preset);
    if (userConfig) {
      return {
        apiKey: userConfig.apiKey,
        baseUrl: userConfig.baseUrl,
        modelName: userConfig.modelName,
      };
    }
  }

  // 2. 按 preset 走环境变量回退链
  switch (preset) {
    case 'chat':
    case 'assistant':
    case 'englishLearning':
      return resolveSiliconFlowCredentials(config);
    case 'knowledgeQa':
      return resolveDashScopeCredentials(config);
    case 'ocr':
      return {
        apiKey: config.get('ZHIPU_API_KEY'),
        baseUrl: config.get('ZHIPU_BASE_URL'),
        modelName: config.get('OCR_MODEL_NAME') || 'glm-4v-flash',
      };
    default:
      return resolveSiliconFlowCredentials(config);
  }
}

function resolveSiliconFlowCredentials(config: ConfigService) {
  const apiKey = config.get('SILICONFLOW_API_KEY');
  if (!apiKey) {
    throw new Error('SILICONFLOW_API_KEY not configured');
  }
  return {
    apiKey,
    baseUrl: config.get('SILICONFLOW_BASE_URL') || 'https://api.siliconflow.cn/v1',
    modelName: config.get('SILICONFLOW_MODEL_NAME') || 'Pro/zai-org/GLM-5.1',
  };
}
```

### 6.3.2 后端持久化

```typescript
// llm-config.service.ts
@Injectable()
export class LlmConfigService {
  async saveConfig(userId: string, config: SaveLlmConfigDto) {
    // API Key AES-256-GCM 加密存储
    const encryptedKey = this.encrypt(config.apiKey);

    const entity = this.configRepo.create({
      userId,
      preset: 'chat',
      apiKey: encryptedKey,
      baseUrl: config.baseUrl,
      modelName: config.modelName,
      embeddingUrl: config.embeddingUrl,
      embeddingModel: config.embeddingModel,
      updatedAt: new Date(),
    });

    await this.configRepo.save(entity);
    return { success: true };
  }

  async getConfig(userId: string) {
    const entity = await this.configRepo.findOne({ where: { userId } });
    if (!entity) return null;

    return {
      ...entity,
      apiKey: entity.apiKey ? this.decrypt(entity.apiKey) : '',
    };
  }

  private encrypt(plaintext: string): string {
    // AES-256-GCM 加密
    const key = createHash('sha256')
      .update(this.configService.get('JWT_SECRET'))
      .digest();
    const cipher = createCipheriv('aes-256-gcm', key, randomBytes(12));
    const encrypted = Buffer.concat([cipher.update(plaintext, 'utf8'), cipher.final()]);
    return Buffer.concat([cipher.getAuthTag(), encrypted]).toString('base64');
  }
}
```

|------|------|------|
| `/settings/llm` | GET | 获取当前用户 LLM 配置 |
