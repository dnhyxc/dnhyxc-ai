## 9.2 词包与经典语句生成

### 9.2.1 主 Agent 检索

```typescript
// english-learning.service.ts
async *streamVocabularyPack(dto: VocabularyPackDto) {
  const { topic, count, enableWebSearch } = dto;

  // 时效策略推断
  const timeStrategy = resolveWebSearchTime(topic, enableWebSearch);

  // 主 Agent 工具决策
  const toolDecision = await this.agentService.decideTools({
    topic,
    timeStrategy,
  });

  // 按需联网检索
  if (toolDecision.useWebSearch) {
    yield { type: 'progress', data: '正在检索最新信息...' };
    const searchResults = await this.serperService.search(topic);
    yield { type: 'agent_tool', data: 'web_search' };
  }

  // 主模型生成
  const prompt = buildVocabularyPrompt(topic, count, toolDecision);
  const llm = createLlm({ preset: 'englishLearning' });
  const stream = await llm.stream([{ role: 'user', content: prompt }]);

  let chunkBuffer = '';
  for await (const chunk of stream) {
    chunkBuffer += chunk.content;

    // 流式解析词条
    const parsed = parseVocabularyChunk(chunkBuffer);
    if (parsed.length) {
      yield { type: 'chunk', data: parsed };
      chunkBuffer = chunkBuffer.slice(parsed.join('').length);
    }
  }

  // 保存结果
  const entries = parseVocabularyChunk(chunkBuffer);
  yield { type: 'complete', data: { total: entries.length } };
}
```

### 9.2.2 时效策略

```typescript
// constant.ts
const TIME_STRATEGY_RULES: Array<{ keywords: RegExp; strategy: TimeStrategy }> = [
  { keywords: /\d{4}年|\d{2}世纪|历史|古代|典籍|GRE|TOEFL/i, strategy: 'none' },
  { keywords: /今天|今日|最近|最新|今年|202[5-9]/i, strategy: 'realtime' },
  { keywords: /周|月|季度|年/i, strategy: 'weekly' },
  { keywords: /昨天|昨日|刚才|刚刚/i, strategy: 'daily' },
  { keywords: /发布|上市|发布于|released/i, strategy: 'explicit_date' },
];

function resolveWebSearchTime(topic: string, enableWebSearch?: boolean): TimeStrategy {
  if (enableWebSearch === false) return 'none';
  if (enableWebSearch === 'true') return 'realtime';

  for (const rule of TIME_STRATEGY_RULES) {
    if (rule.keywords.test(topic)) return rule.strategy;
  }

  // 默认：启用联网搜索
  return 'realtime';
}
```

## 9.3 Agent 多会话

### 9.3.1 多会话隔离

```typescript
// store/englishAgent.ts
class EnglishAgentStore {
  activeSessionId: string | null = null;
  stateBySession = new Map<string, SessionState>();
  historySessions: SessionMeta[] = [];

  setActiveSession(sessionId: string) {
    this.activeSessionId = sessionId;
  }

  async loadHistory(page = 1) {
    const res = await api.get('/agent/sessions', {
      params: { page, pageSize: 20 },
    });
    this.historySessions = res.list;
  }

  async createSession() {
    const sessionId = `session-${Date.now()}`;
    this.activeSessionId = sessionId;
    this.stateBySession.set(sessionId, {
      messages: [],
      isStreaming: false,
      streamId: null,
    });
    return sessionId;
  }

  async sendMessage(content: string) {
    const sessionId = this.activeSessionId || await this.createSession();
    const state = this.stateBySession.get(sessionId);

    state.isStreaming = true;

    // SSE 流式
    for await (const event of agentSse(sessionId, content)) {
      if (event.type === 'thinking') {
        state.messages.push({ type: 'thinking', content: event.data });
      } else if (event.type === 'content') {
        const last = state.messages[state.messages.length - 1];
        if (last?.type === 'content') {
          last.content += event.data;
        } else {
          state.messages.push({ type: 'content', content: event.data });
        }
      } else if (event.type === 'meta') {
        state.streamId = event.data.streamId;
      }
    }

    state.isStreaming = false;
  }
}
```

### 9.4.1 播放选路

```typescript
// utils/speech.ts
export async function playSpeech(
  text: string,
  config: SpeechConfig,
): Promise<void> {
  const { playbackSource, isMember } = config;

  if (playbackSource === 'cloud' && isMember) {
    return playCloudTts(text, config);
  }
  return playLocalTts(text, config);
}

async function playCloudTts(text: string, config: SpeechConfig) {
  try {
    // 优先 MiniMax
    return playMiniMaxTts(text, config);
  } catch (e) {
    // MiniMax 不可用 → 回退硅基
    if (isMiniMaxError(e)) {
      try {
        return playSiliconFlowTts(text, config);
      } catch {
        // 最终回退本机
        return playLocalTts(text, config);
      }
    }
    throw e;
  }
}

function playLocalTts(text: string, config: SpeechConfig) {
  const utterance = new SpeechSynthesisUtterance(text);
  utterance.lang = config.language || 'en-US';
  utterance.rate = config.rate || 1.0;
  utterance.pitch = config.pitch || 1.0;
  utterance.volume = config.volume || 1.0;

  if (config.voice) {
    const voices = speechSynthesis.getVoices();
    const voice = voices.find(v => v.name === config.voice);
    if (voice) utterance.voice = voice;
  }

  speechSynthesis.speak(utterance);
  return new Promise<void>((resolve) => {
    utterance.onend = () => resolve();
    utterance.onerror = () => resolve();
  });
}
```

### 9.4.2 MiniMax 流式 TTS

```typescript
async function playMiniMaxTts(text: string, config: SpeechConfig) {
  const response = await fetch(
    '/speech-transcription/minimax/speech/stream',
    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${config.apiKey}`,
      },
      body: JSON.stringify({
        text,
        model: config.model || 'speech-2.8-turbo',
        voice_id: config.voice,
        speed: config.rate,
        volume: config.volume,
        pitch: config.pitch,
        format: 'pcm',
        sample_rate: 24000,
      }),
    },
  );

  if (!response.ok) throw new Error(`TTS failed: ${response.status}`);

  // hex 音频逐块转发
  const reader = response.body!.getReader();
  const chunks: Uint8Array[] = [];

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    chunks.push(value);
  }

  // 播放
  const audioBlob = new Blob(chunks, { type: 'audio/pcm' });
  const url = URL.createObjectURL(audioBlob);
  const audio = new Audio(url);
  await audio.play();
  audio.onended = () => URL.revokeObjectURL(url);
}
```

```typescript
// english-practice-review.srs.ts
export function calculateSrsSchedule(
  isCorrect: boolean,
  currentEase: number,
  currentInterval: number,
  repetitions: number,
): { nextInterval: number; nextEase: number } {
  let ease = currentEase;
  let interval = currentInterval;

  if (isCorrect) {
    repetitions++;
    if (repetitions === 1) {
      interval = 1;  // 明天
    } else if (repetitions === 2) {
      interval = 3;  // 3 天后
    } else {
      interval = Math.round(interval * ease);
    }
    ease = Math.min(ease + 0.1, 3.0);
  } else {
    repetitions = 0;
    interval = 1;  // 答错明天复习
    ease = Math.max(ease - 0.2, 1.3);
  }

  return { nextInterval: interval, nextEase: ease };
}
```

```typescript
class LearningNotesStore {
  notes: LearningNote[] = [];
  currentNote: LearningNote | null = null;
  isDirty = false;

  async loadNotes(page = 1) {
    const res = await api.get('/english-learning/notes', { params: { page } });
    this.notes = res.list;
  }

  async createNote(title: string, content: string) {
    const note = await api.post('/english-learning/notes', { title, content });
    this.notes.unshift(note);
    this.currentNote = note;
  }

  async updateNote(id: string, content: string) {
    const note = await api.put(`/english-learning/notes/${id}`, { content });
    const idx = this.notes.findIndex(n => n.id === id);
    if (idx >= 0) this.notes[idx] = note;
    this.isDirty = false;
  }

  async deleteNote(id: string) {
    await api.delete(`/english-learning/notes/${id}`);
    this.notes = this.notes.filter(n => n.id !== id);
    if (this.currentNote?.id === id) this.currentNote = null;
  }
}
```

### 9.8.2 虚拟网格：行级虚拟化 + 自适应列

```typescript
// hooks/useLibraryGridColumns.ts
import { useEffect, useState } from 'react';

export function useLibraryGridColumns() {
  const [columns, setColumns] = useState(2);

  useEffect(() => {
    const calc = () => {
      const w = window.innerWidth;
      if (w <= 640) setColumns(1);        // 窄屏：1 列
      else if (w <= 1024) setColumns(2);  // 中屏：2 列
      else setColumns(3);                 // 宽屏：3 列
    };
    calc();
    window.addEventListener('resize', calc);
    return () => window.removeEventListener('resize', calc);
  }, []);

  return columns;
}

// components/LibraryVirtuosoGrid.tsx（核心骨架）
import { Virtuoso } from 'react-virtuoso';
import { chunk } from 'lodash-es';

type Props<T> = {
  items: T[];
  initialScrollItemIndex?: number;  // 词条级 index，不是行号
  renderCard: (item: T, i: number) => React.ReactNode;
  onEndReached?: () => void;
};

export function LibraryVirtuosoGrid<T extends { id: string }>(props: Props<T>) {
  const { items, initialScrollItemIndex = 0, renderCard, onEndReached } = props;
  const columns = useLibraryGridColumns();
  // 词条数组按列数切成「行」
  const rows = chunk(items, columns);
  // 行号 = 向上取整 (词条 index / columns)
  const initialRowIndex = Math.ceil(initialScrollItemIndex / columns);

  return (
    <Virtuoso
      data={rows}
      initialTopMostItemIndex={initialRowIndex}
      endReached={onEndReached}
      itemContent={(_, row) => (
        <div
          style={{
            display: 'grid',
            gridTemplateColumns: `repeat(${columns}, minmax(0, 1fr))`,
            gap: 16,
          }}
        >
          {row.map((item, rowCol) => {
            const flatIndex = Math.floor(items.indexOf(item) / 1) * 0; // 忽略占位
            return renderCard(item, flatIndex);
          })}
        </div>
      )}
    />
  );
}
```

### 9.8.3 单向滚动 + 并发预取（核心 Hook）

```typescript
// hooks/useLibraryWordsList.ts
const LIBRARY_ITEMS_FETCH_CHUNK = 1000;  // 每块最大 1000 条（后端 limit 上限）
const PREFETCH_CONCURRENCY = 3;          // 并发拉取 worker 数

// 分块：把窗口 [0, resumeOffset+pageSize) 切成 limit=1000 的若干片
function buildLibraryPrefetchChunks(resumeOffset: number, pageSize: number) {
  const end = Math.max(resumeOffset + pageSize, pageSize);
  const chunks: { offset: number; limit: number }[] = [];
  for (let off = 0; off < end; off += LIBRARY_ITEMS_FETCH_CHUNK) {
    chunks.push({
      offset: off,
      limit: Math.min(LIBRARY_ITEMS_FETCH_CHUNK, end - off),
    });
  }
  return chunks;
}

// 并发池：开 workers 个「取任务 → 执行 → 取下一个」线程，结果顺序 = 输入顺序
async function mapWithConcurrency<A, B>(
  tasks: A[],
  workers: number,
  runner: (a: A) => Promise<B>,
): Promise<B[]> {
  const out: B[] = new Array(tasks.length);
  let next = 0;
  await Promise.all(
    Array.from({ length: workers }, async () => {
      while (true) {
        const i = next++;
        if (i >= tasks.length) return;
        out[i] = await runner(tasks[i]);
      }
    }),
  );
  return out;
}

// 首屏：并发预取完整窗口 + 滚动锚点对齐
const fetchInitialWindow = useCallback(
  async (id: string, gen: number, resumeOffset: number) => {
    setLoading(true);
    try {
      const chunks = buildLibraryPrefetchChunks(resumeOffset, pageSize);
      const pages = await mapWithConcurrency(
        chunks,
        PREFETCH_CONCURRENCY,
        (c) => fetchPageWithRetry(id, c.limit, c.offset),
      );
      // 顺序合并 pages[0].items + pages[1].items …
      const list = pages.flatMap((p) => (Array.isArray(p.items) ? p.items : []));
      const scrollIdx = Math.min(Math.max(resumeOffset, 0), Math.max(list.length - 1, 0));
      applyWindow(id, 0, list, pages[0]?.library, scrollIdx, true);
    } catch {
      Toast({ type: 'error', title: '词条加载失败' });
    } finally {
      if (gen === loadGenRef.current) setLoading(false);
    }
  },
  [applyWindow, fetchPageWithRetry, pageSize],
);

// 缓存恢复：切库时先短路命中本地 libraryWordsListCache
useEffect(() => {
  if (!libraryId) return;
  const gen = ++loadGenRef.current;
  const resume = alignResumeOffset(initialResumeOffset ?? 0, pageSize);
  if (restoreFromCache(libraryId, resume)) return; // 命中：秒开
  void fetchInitialWindow(libraryId, gen, resume);
}, [libraryId, fetchInitialWindow, restoreFromCache, initialResumeOffset, pageSize]);

// 向下加载更多（单向：仅 append）；守卫 listReady/hasMore/fetchingMore/lastMoreOffset
const fetchMore = useCallback(async () => {
  if (!listReadyRef.current || !hasMoreRef.current || fetchingMoreRef.current) return;
  const offset = endOffsetRef.current;
  if (lastMoreOffsetRef.current === offset) return; // 去重
  fetchingMoreRef.current = true;
  lastMoreOffsetRef.current = offset;
  setLoadingMore(true);
  try {
    const data = await fetchPageWithRetry(libraryIdRef.current!, pageSize, offset);
    setItems((prev) => {
      const next = [...prev, ...(Array.isArray(data.items) ? data.items : [])];
      endOffsetRef.current = next.length;
      hasMoreRef.current = data.items?.length >= pageSize;
      return next;
    });
  } finally {
    fetchingMoreRef.current = false;
    lastMoreOffsetRef.current = -1;
    setLoadingMore(false);
  }
}, [fetchPageWithRetry, pageSize]);
```

### 9.8.5 续读持久化：按页对齐 + keepalive flush

```typescript
// 续读写入：先按页对齐（避免每次滚动都发请求），相同 offset 直接跳过
function alignResumeOffset(offset: number, pageSize: number) {
  return Math.max(0, Math.floor(offset / pageSize) * pageSize);
}

const setResume = useCallback(
  (offset: number) => {
    const aligned = alignResumeOffset(offset, pageSize);
    if (aligned === lastSavedResumeRef.current) return;
    lastSavedResumeRef.current = aligned;
    onResumeOffsetChange?.(aligned);  // 外层调用方的 PUT
  },
  [onResumeOffsetChange, pageSize],
);

// 离开页面/切后台：用 keepalive 确保请求发出（即使关标签页）
useEffect(() => {
  const flush = () => {
    const off = lastSavedResumeRef.current;
    if (off < 0 || !persistResume || !libraryIdRef.current) return;
    fetch(`/english-learning/vocabulary-libraries/${libraryIdRef.current}/resume?offset=${off}`, {
      method: 'PUT',
      keepalive: true,
      headers: { Authorization: `Bearer ${token}` },
    }).catch(() => void 0);
  };
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) flush();
  });
  window.addEventListener('beforeunload', flush);
  return () => {
    window.removeEventListener('beforeunload', flush);
  };
}, [persistResume]);
```
