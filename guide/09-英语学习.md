## 9.2 词包与经典语句生成

### 9.2.1 主 Agent 检索

```typescript
// english-learning.service.ts
async *streamVocabularyPack(dto: VocabularyPackDto) {
  const { topic, count, enableWebSearch } = dto;

  // 时效策略推断
  const timeStrategy = resolveWebSearchTime(topic, enableWebSearch);

  // 主 Agent 工具决策
  const toolDecision = await this.agentService.decideTools({
    topic,
    timeStrategy,
  });

  // 按需联网检索
  if (toolDecision.useWebSearch) {
    yield { type: 'progress', data: '正在检索最新信息...' };
    const searchResults = await this.serperService.search(topic);
    yield { type: 'agent_tool', data: 'web_search' };
  }

  // 主模型生成
  const prompt = buildVocabularyPrompt(topic, count, toolDecision);
  const llm = createLlm({ preset: 'englishLearning' });
  const stream = await llm.stream([{ role: 'user', content: prompt }]);

  let chunkBuffer = '';
  for await (const chunk of stream) {
    chunkBuffer += chunk.content;

    // 流式解析词条
    const parsed = parseVocabularyChunk(chunkBuffer);
    if (parsed.length) {
      yield { type: 'chunk', data: parsed };
      chunkBuffer = chunkBuffer.slice(parsed.join('').length);
    }
  }

  // 保存结果
  const entries = parseVocabularyChunk(chunkBuffer);
  yield { type: 'complete', data: { total: entries.length } };
}
```

### 9.2.2 时效策略

```typescript
// constant.ts
const TIME_STRATEGY_RULES: Array<{ keywords: RegExp; strategy: TimeStrategy }> = [
  { keywords: /\d{4}年|\d{2}世纪|历史|古代|典籍|GRE|TOEFL/i, strategy: 'none' },
  { keywords: /今天|今日|最近|最新|今年|202[5-9]/i, strategy: 'realtime' },
  { keywords: /周|月|季度|年/i, strategy: 'weekly' },
  { keywords: /昨天|昨日|刚才|刚刚/i, strategy: 'daily' },
  { keywords: /发布|上市|发布于|released/i, strategy: 'explicit_date' },
];

function resolveWebSearchTime(topic: string, enableWebSearch?: boolean): TimeStrategy {
  if (enableWebSearch === false) return 'none';
  if (enableWebSearch === 'true') return 'realtime';

  for (const rule of TIME_STRATEGY_RULES) {
    if (rule.keywords.test(topic)) return rule.strategy;
  }

  // 默认：启用联网搜索
  return 'realtime';
}
```

## 9.3 Agent 多会话

### 9.3.1 多会话隔离

```typescript
// store/englishAgent.ts
class EnglishAgentStore {
  activeSessionId: string | null = null;
  stateBySession = new Map<string, SessionState>();
  historySessions: SessionMeta[] = [];

  setActiveSession(sessionId: string) {
    this.activeSessionId = sessionId;
  }

  async loadHistory(page = 1) {
    const res = await api.get('/agent/sessions', {
      params: { page, pageSize: 20 },
    });
    this.historySessions = res.list;
  }

  async createSession() {
    const sessionId = `session-${Date.now()}`;
    this.activeSessionId = sessionId;
    this.stateBySession.set(sessionId, {
      messages: [],
      isStreaming: false,
      streamId: null,
    });
    return sessionId;
  }

  async sendMessage(content: string) {
    const sessionId = this.activeSessionId || await this.createSession();
    const state = this.stateBySession.get(sessionId);

    state.isStreaming = true;

    // SSE 流式
    for await (const event of agentSse(sessionId, content)) {
      if (event.type === 'thinking') {
        state.messages.push({ type: 'thinking', content: event.data });
      } else if (event.type === 'content') {
        const last = state.messages[state.messages.length - 1];
        if (last?.type === 'content') {
          last.content += event.data;
        } else {
          state.messages.push({ type: 'content', content: event.data });
        }
      } else if (event.type === 'meta') {
        state.streamId = event.data.streamId;
      }
    }

    state.isStreaming = false;
  }
}
```

### 9.4.1 播放选路

```typescript
// utils/speech.ts
export async function playSpeech(
  text: string,
  config: SpeechConfig,
): Promise<void> {
  const { playbackSource, isMember } = config;

  if (playbackSource === 'cloud' && isMember) {
    return playCloudTts(text, config);
  }
  return playLocalTts(text, config);
}

async function playCloudTts(text: string, config: SpeechConfig) {
  try {
    // 优先 MiniMax
    return playMiniMaxTts(text, config);
  } catch (e) {
    // MiniMax 不可用 → 回退硅基
    if (isMiniMaxError(e)) {
      try {
        return playSiliconFlowTts(text, config);
      } catch {
        // 最终回退本机
        return playLocalTts(text, config);
      }
    }
    throw e;
  }
}

function playLocalTts(text: string, config: SpeechConfig) {
  const utterance = new SpeechSynthesisUtterance(text);
  utterance.lang = config.language || 'en-US';
  utterance.rate = config.rate || 1.0;
  utterance.pitch = config.pitch || 1.0;
  utterance.volume = config.volume || 1.0;

  if (config.voice) {
    const voices = speechSynthesis.getVoices();
    const voice = voices.find(v => v.name === config.voice);
    if (voice) utterance.voice = voice;
  }

  speechSynthesis.speak(utterance);
  return new Promise<void>((resolve) => {
    utterance.onend = () => resolve();
    utterance.onerror = () => resolve();
  });
}
```

### 9.4.2 MiniMax 流式 TTS

```typescript
async function playMiniMaxTts(text: string, config: SpeechConfig) {
  const response = await fetch(
    '/speech-transcription/minimax/speech/stream',
    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${config.apiKey}`,
      },
      body: JSON.stringify({
        text,
        model: config.model || 'speech-2.8-turbo',
        voice_id: config.voice,
        speed: config.rate,
        volume: config.volume,
        pitch: config.pitch,
        format: 'pcm',
        sample_rate: 24000,
      }),
    },
  );

  if (!response.ok) throw new Error(`TTS failed: ${response.status}`);

  // hex 音频逐块转发
  const reader = response.body!.getReader();
  const chunks: Uint8Array[] = [];

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    chunks.push(value);
  }

  // 播放
  const audioBlob = new Blob(chunks, { type: 'audio/pcm' });
  const url = URL.createObjectURL(audioBlob);
  const audio = new Audio(url);
  await audio.play();
  audio.onended = () => URL.revokeObjectURL(url);
}
```

```typescript
// english-practice-review.srs.ts
export function calculateSrsSchedule(
  isCorrect: boolean,
  currentEase: number,
  currentInterval: number,
  repetitions: number,
): { nextInterval: number; nextEase: number } {
  let ease = currentEase;
  let interval = currentInterval;

  if (isCorrect) {
    repetitions++;
    if (repetitions === 1) {
      interval = 1;  // 明天
    } else if (repetitions === 2) {
      interval = 3;  // 3 天后
    } else {
      interval = Math.round(interval * ease);
    }
    ease = Math.min(ease + 0.1, 3.0);
  } else {
    repetitions = 0;
    interval = 1;  // 答错明天复习
    ease = Math.max(ease - 0.2, 1.3);
  }

  return { nextInterval: interval, nextEase: ease };
}
```

class LearningNotesStore {
  notes: LearningNote[] = [];
  currentNote: LearningNote | null = null;
  isDirty = false;

  async loadNotes(page = 1) {
    const res = await api.get('/english-learning/notes', { params: { page } });
    this.notes = res.list;
  }

  async createNote(title: string, content: string) {
    const note = await api.post('/english-learning/notes', { title, content });
    this.notes.unshift(note);
    this.currentNote = note;
  }

  async updateNote(id: string, content: string) {
    const note = await api.put(`/english-learning/notes/${id}`, { content });
    const idx = this.notes.findIndex(n => n.id === id);
    if (idx >= 0) this.notes[idx] = note;
    this.isDirty = false;
  }

  async deleteNote(id: string) {
    await api.delete(`/english-learning/notes/${id}`);
    this.notes = this.notes.filter(n => n.id !== id);
    if (this.currentNote?.id === id) this.currentNote = null;
  }
}

### 9.8.2 虚拟网格：行级虚拟化 + 自适应列

// hooks/useLibraryGridColumns.ts
import { useEffect, useState } from 'react';

export function useLibraryGridColumns() {
  const [columns, setColumns] = useState(2);

  useEffect(() => {
    const calc = () => {
      const w = window.innerWidth;
      if (w <= 640) setColumns(1);        // 窄屏：1 列
      else if (w <= 1024) setColumns(2);  // 中屏：2 列
      else setColumns(3);                 // 宽屏：3 列
    };
    calc();
    window.addEventListener('resize', calc);
    return () => window.removeEventListener('resize', calc);
  }, []);

  return columns;
}

### 9.8.3 单向滚动 + 并发预取（核心 Hook）

```typescript
// hooks/useLibraryWordsList.ts
const LIBRARY_ITEMS_FETCH_CHUNK = 1000;  // 每块最大 1000 条（后端 limit 上限）
const PREFETCH_CONCURRENCY = 3;          // 并发拉取 worker 数

// 分块：把窗口 [0, resumeOffset+pageSize) 切成 limit=1000 的若干片
function buildLibraryPrefetchChunks(resumeOffset: number, pageSize: number) {
  const end = Math.max(resumeOffset + pageSize, pageSize);
  const chunks: { offset: number; limit: number }[] = [];
  for (let off = 0; off < end; off += LIBRARY_ITEMS_FETCH_CHUNK) {
    chunks.push({
      offset: off,
      limit: Math.min(LIBRARY_ITEMS_FETCH_CHUNK, end - off),
    });
  }
  return chunks;
}

// 并发池：开 workers 个「取任务 → 执行 → 取下一个」线程，结果顺序 = 输入顺序
async function mapWithConcurrency<A, B>(
  tasks: A[],
  workers: number,
  runner: (a: A) => Promise<B>,
): Promise<B[]> {
  const out: B[] = new Array(tasks.length);
  let next = 0;
  await Promise.all(
    Array.from({ length: workers }, async () => {
      while (true) {
        const i = next++;
        if (i >= tasks.length) return;
        out[i] = await runner(tasks[i]);
      }
    }),
  );
  return out;
}

// 首屏：并发预取完整窗口 + 滚动锚点对齐
const fetchInitialWindow = useCallback(
  async (id: string, gen: number, resumeOffset: number) => {
    setLoading(true);
    try {
      const chunks = buildLibraryPrefetchChunks(resumeOffset, pageSize);
      const pages = await mapWithConcurrency(
        chunks,
        PREFETCH_CONCURRENCY,
        (c) => fetchPageWithRetry(id, c.limit, c.offset),
      );
      // 顺序合并 pages[0].items + pages[1].items …
      const list = pages.flatMap((p) => (Array.isArray(p.items) ? p.items : []));
      const scrollIdx = Math.min(Math.max(resumeOffset, 0), Math.max(list.length - 1, 0));
      applyWindow(id, 0, list, pages[0]?.library, scrollIdx, true);
    } catch {
      Toast({ type: 'error', title: '词条加载失败' });
    } finally {
      if (gen === loadGenRef.current) setLoading(false);
    }
  },
  [applyWindow, fetchPageWithRetry, pageSize],
);

// 缓存恢复：切库时先短路命中本地 libraryWordsListCache
useEffect(() => {
  if (!libraryId) return;
  const gen = ++loadGenRef.current;
  const resume = alignResumeOffset(initialResumeOffset ?? 0, pageSize);
  if (restoreFromCache(libraryId, resume)) return; // 命中：秒开
  void fetchInitialWindow(libraryId, gen, resume);
}, [libraryId, fetchInitialWindow, restoreFromCache, initialResumeOffset, pageSize]);

// 向下加载更多（单向：仅 append）；守卫 listReady/hasMore/fetchingMore/lastMoreOffset
const fetchMore = useCallback(async () => {
  if (!listReadyRef.current || !hasMoreRef.current || fetchingMoreRef.current) return;
  const offset = endOffsetRef.current;
  if (lastMoreOffsetRef.current === offset) return; // 去重
  fetchingMoreRef.current = true;
  lastMoreOffsetRef.current = offset;
  setLoadingMore(true);
  try {
    const data = await fetchPageWithRetry(libraryIdRef.current!, pageSize, offset);
    setItems((prev) => {
      const next = [...prev, ...(Array.isArray(data.items) ? data.items : [])];
      endOffsetRef.current = next.length;
      hasMoreRef.current = data.items?.length >= pageSize;
      return next;
    });
  } finally {
    fetchingMoreRef.current = false;
    lastMoreOffsetRef.current = -1;
    setLoadingMore(false);
  }
}, [fetchPageWithRetry, pageSize]);
```

## 9.10 列表悬浮角标

### 9.10.1 背景

收藏、错题集、每日记词记录等列表改用虚拟滚动后，长列表无法用原生滚动条快速跳转顶/底。本次新增「列表悬浮角标」组件系统：右下角浮动按钮，可滚动时显示「置底」，触底后切换为「置顶」。核心难点是 FAB 锚点停靠——点击「置底」后按钮停在加载区上方（避免立即触发 `endReached`），用户手动滚入加载区后释放锚点恢复常规触底判定。

### 9.10.2 架构

```

│    ├─ useListScrollCornerFab(viewportRef, items.length)     │
│    │    ├─ mode: 'hidden' | 'toBottom' | 'toTop'           │
│    │    ├─ onScrollCornerFab → 滚动事件中更新 mode          │
│    │    └─ onScrollCornerFabClick → 点击 FAB 执行滚动       │
│    ├─ ListScrollCornerFab { mode, onClick }                 │
│    └─ composeViewportScroll(onViewportScroll, onScrollCornerFab) │
│                                                              │
│  LibraryVirtuosoGrid 改造：                                   │
│    ├─ atBottomThreshold = LIBRARY_LIST_AT_BOTTOM_THRESHOLD_PX │
│    ├─ endReached → shouldBlockLibraryListEndReached(vp) 拦截 │
│    └─ registerLibraryVirtuosoScrollParent(scrollParent, handle) │

## 9.11 续读模块开关

## 9.12 收藏状态会话级缓存

### 9.12.1 背景

function readVocabFavorited(word, embeddedFavoriteId, favoriteIdByWordKey): boolean {
  const wk = normalizeEnglishVocabWordKey(word);
  if (!wk) return false;
  const sessionId = sessionFavoriteIdByWordKey.get(wk);

// views/englishLearning/utils/libraryWordsListCache.ts — 列表快照 patch
export function patchVocabFavoriteInListCaches(wordKey: string, favoriteId: string | null) {
  const wk = wordKey.trim().toLowerCase();
  if (!wk) return;
  // 遍历所有列表 entry，命中 wordKey 的 item 同步 favoriteId 字段并 touch LRU
  for (const [key, slot] of store) {
    const entry = slot.data as LibraryWordsListCacheEntry<
      { word: string; favoriteId?: string | null },
      unknown
    >;
    let changed = false;
    const items = entry.items.map((item) => {
      if (normalizeEnglishVocabWordKey(item.word) !== wk) return item;
      const next = favoriteId;
      if ((item.favoriteId ?? null) === next) return item;
      changed = true;
      return { ...item, favoriteId: next };
    });
    if (changed) {
      entry.items = items;
      touchLru(key, slot);  // 提升命中 entry 的 LRU 优先级
    }
  }
}

private async attachVocabularyFavoriteIdsByWord<T extends { word: string }>(
  userId: number,
  items: T[],
): Promise<Array<T & { favoriteId: string | null }>> {
  if (items.length === 0) return [];
  // 批量查当前用户对这些 word 的收藏记录
  const refs = await this.listVocabularyFavoriteRefsForWords(
    userId,
    items.map((i) => i.word),

  // 直接在响应体上挂 favoriteId 字段，前端无需二次 status 查询
  return items.map((item) => {
    const wordKey = this.normalizeVocabularyFavoriteWordKey(item.word);
    const favoriteId = wordKey ? (idByWordKey.get(wordKey) ?? null) : null;
    return { ...item, favoriteId };
  });
}

function useFavoriteToggleBindings(props: FavoriteToggleButtonProps) {
  const { t } = useI18n();
  const { kind } = props;
  const vocabItem = kind === 'vocab' ? props.item : null;
  const classicItem = kind === 'classic' ? props.item : null;

  // 单元素数组喂给增量 Hook，仅用于复用其状态读取 / 设置能力
  const vocabCardItems = useMemo(() => (vocabItem ? [{ word: vocabItem.word }] : []), [vocabItem?.word]);
  const classicCardItems = useMemo(() => (classicItem ? [{ english: classicItem.english }] : []), [classicItem?.english]);

  const vocabStatus = useIncrementalVocabFavoriteStatus(vocabCardItems);
  const classicStatus = useIncrementalClassicQuoteFavoriteStatus(classicCardItems);

  // 统一 itemKey：vocab 用 wordKey，classic 用 contentKey
  const itemKey = kind === 'vocab' && vocabItem
    ? normalizeEnglishVocabWordKey(vocabItem.word)
    : classicItem ? classicQuoteFavoriteContentKey(classicItem.english) : '';

  // 当前是否收藏（按 kind 路由到对应 Hook 的 isXxxFavorited）
  const isFavorited = kind === 'vocab' && vocabItem
    ? vocabStatus.isVocabularyFavorited(vocabItem.word, vocabItem.favoriteId)
    : classicItem ? classicStatus.isClassicQuoteFavorited(classicItem.english, classicItem.favoriteId) : false;

  const [favBusy, setFavBusy] = useState(false);

  // toggle 主流程：busy 守卫 → 已收藏则 remove + clear，未收藏则 add + set
  const onToggle = useCallback(async () => {
    if (!itemKey || favBusy) return;
    setFavBusy(true);
    try {
      if (checkFavorited()) {
        const favoriteId = resolveFavoriteId();
        if (!favoriteId) { clearFavorite(); return; }
        await removeFavorite(favoriteId);
        clearFavorite();
      } else {
        const newId = await addFavorite();
        if (newId) setFavoriteId(newId);
      }
    } catch {
      // 错误提示由 http 客户端统一处理
    } finally {
      setFavBusy(false);
    }
  }, [addFavorite, checkFavorited, clearFavorite, favBusy, itemKey, removeFavorite, resolveFavoriteId, setFavoriteId]);

  return { favBusy, isFavorited, onToggle, favoriteLabel, unfavoriteLabel };
}

## 9.14 学习笔记图片上传与回收

### 9.14.1 背景

学习笔记富文本编辑器（TipTap）此前存在两条分裂的图片路径：

- **Web 浏览器粘贴 / 拖拽**：TipTap Image 扩展自带 `onUploadImage` 钩子，前端拿到 `File` 后 POST 到后端，后端上传 COS 返回 `notes/...` URL，编辑器回填 `<img src="https://.../notes/...">`。这条路径此前已可用。
- **Tauri 桌面端粘贴**：剪贴板里的位图、文件列表、富文本 HTML 在 `apps/frontend/src/utils/clipboard.ts` 的 `attachTauriPlainFieldClipboardShortcuts` 中被读成 `data:image/...;base64,...`，旧实现**直接以 `<img src="data:...">` 插入 ProseMirror 文档**。后果：①笔记正文塞满 base64，几张截图就内存溢出 / 序列化卡顿；②DOCX 导出（`learning-note-docx.builder.ts`）必须对 base64 做 `atob` + `sharp` 转码，长笔记极易超时；③`EnglishLearningNote.content` 列（`text` 5MB 上限）很快撑爆；④桌面端与 Web 端图片处理路径分裂。

即便走通 `onUploadImage` 的图片也存在「孤儿对象」问题：

- 用户新建笔记先粘贴若干图（前端调 `POST .../upload-image` 已写入 COS），随后**关闭或刷新而没保存**：COS 上多出几条永远不被任何笔记引用的图（孤儿）。
- 用户编辑已有笔记先粘贴新图（已上传 COS），随后**把图删掉再保存**：旧正文有该图、新正文没有，COS 上残留孤儿。
- 用户**删整篇笔记**：COS 上对应的图若没有引用计数回收，永远残留。

本次目标：把笔记正文图片统一为「服务端 COS 上传 + 附件表引用计数 + pending 会话结算 + 孤儿自动回收」全链路；桌面端粘贴的 `data:image/` 也转 File 后派发 `rich-editor:desktop-paste-images` 自定义事件，由 RichEditor ImageUpload 扩展 `preventDefault` 后走 `onUploadImage` 钩子，与本后端接口协同。

async uploadImage(userId, file, noteId?, sessionId?) {
  // 防御：没文件 / 文件空 buffer
  if (!file?.buffer?.length) throw new BadRequestException('请上传图片文件');
  // 防御：MIME 非图片（与 controller fileFilter 双层保险）
  if (!file.mimetype?.startsWith('image/')) throw new BadRequestException('仅支持图片文件');
  // 调 UploadService 上传 COS，前缀 notes/（buildCosObjectKey 自动生成 notes/{uuid}_name）
  const uploaded = await this.uploadService.uploadObjectToCos(file, 'notes');
  const sid = sessionId?.trim();
  const nid = noteId?.trim();
  // 有会话 ID：登记到 pending 表（保存 / 放弃 / settle 时再处理）
  if (sid) await this.ensurePendingRow(userId, sid, uploaded.key, uploaded.url);
  // 仅笔记 ID：登记到 attachment 表（已有笔记直接落附件行）
  else if (nid) await this.ensureAttachmentRow(userId, nid, uploaded.key, uploaded.url);
  // 顺手 GC 该用户过期 pending（任意写入口都触发兜底）
  void this.gcExpiredPending(userId);
  return uploaded; // { key, url } 给前端回填 <img src>
}

// learning-notes.service.ts — syncAttachmentsFromContent（正文为真相源 diff 附件表）
private async syncAttachmentsFromContent(noteId, html) {
  // 抽出正文所有 notes/ 引用
  const refs = extractNoteImageRefsFromHtml(html);
  // 当前正文 key 集合
  const nextKeys = new Set(refs.map(r => r.key));
  // 查附件表现有行
  const existing = await this.attachmentRepo.find({ where: { noteId }, select: { id: true, cosKey: true } });
  const existingKeys = new Set(existing.map(a => a.cosKey));

  // 现有但正文已无的 → 删行
  const toRemove = existing.filter(a => !nextKeys.has(a.cosKey));
  if (toRemove.length) await this.attachmentRepo.delete({ id: In(toRemove.map(a => a.id)) });

  // 正文有但现有附件表无的 → 增行（url 缺失时由 buildCosPublicUrl 重建）
  const toAdd = refs.filter(r => !existingKeys.has(r.key));
  if (toAdd.length) {
    await this.attachmentRepo.save(toAdd.map(r => this.attachmentRepo.create({
      noteId, cosKey: r.key,
      url: r.url || this.uploadService.buildCosPublicUrl(r.key),
    })));
  }
  // 本篇移除的 key 走引用计数回收（pending 表 / 其它笔记仍引用则跳过）
  await this.deleteOrphanCosKeys(toRemove.map(a => a.cosKey));
}
```

// learning-notes.service.ts — settlePendingSession（保存时结算会话）
private async settlePendingSession(userId, sessionId, html) {
  // 抽出当前正文里所有 notes/ key（引用计数白名单）
  const kept = new Set(extractNoteImageRefsFromHtml(html).map(r => r.key));
  // 查该会话所有 pending 行
  const rows = await this.pendingRepo.find({ where: { userId, sessionId }, select: { id: true, cosKey: true } });
  if (!rows.length) return;

  // 拆分：正文里没有的 key 视为孤儿
  const orphanKeys = rows.filter(r => !kept.has(r.cosKey)).map(r => r.cosKey);
  // 先删所有 pending 行（无论是否孤儿）
  await this.pendingRepo.delete({ id: In(rows.map(r => r.id)) });
  // 再回收孤儿 key（attachment 表若无引用即可删 COS）
  await this.deleteOrphanCosKeys(orphanKeys);
}

// learning-notes.service.ts — deleteOrphanCosKeys（双表 count 引用计数回收）
private async deleteOrphanCosKeys(keys: string[]) {
  // 过滤 + 去重：仅 notes/ 域 key 才回收（防误删 assets / ebooks 等他域）
  const unique = [...new Set(keys.filter(k => k.startsWith('notes/')))];
  await Promise.all(unique.map(async key => {
    try {
      // 双表 count：attachment 表 + pending 表
      const [inNote, inPending] = await Promise.all([
        this.attachmentRepo.count({ where: { cosKey: key } }),
        this.pendingRepo.count({ where: { cosKey: key } }),
      ]);
      // 任一表仍引用则跳过
      if (inNote > 0 || inPending > 0) return;
      // 真正孤儿：删 COS 对象
      await this.uploadService.deleteCosObject(key);
    } catch (e) {
      // 单 key 回收失败不阻断其它 key
      this.logger.warn(`回收笔记图片失败 key=${key}: ${e instanceof Error ? e.message : e}`);
    }
  }));
}

// learning-notes.service.ts — gcExpiredPending（TTL 兜底，崩溃会话回收）
private async gcExpiredPending(userId) {
  try {
    // 截断时间点：当前时间往前 1h
    const cutoff = new Date(Date.now() - PENDING_TTL_MS);
    // 查该用户 1h 前的 pending（最多 100 条，避免单次 GC 拖慢主请求）
    const rows = await this.pendingRepo.find({
      where: { userId, createdAt: LessThan(cutoff) },
      select: { id: true, cosKey: true },
      take: 100,
    });
    if (!rows.length) return;
    const keys = rows.map(r => r.cosKey);
    await this.pendingRepo.delete({ id: In(rows.map(r => r.id)) });
    await this.deleteOrphanCosKeys(keys);
  } catch (e) {
    // GC 失败不阻断主请求，仅 warn
    this.logger.warn(`清理过期 pending 失败: ${e instanceof Error ? e.message : e}`);
  }
}

// frontend/utils/clipboard.ts — dataUrlToFile + offerDesktopPasteImages
function dataUrlToFile(dataUrl, index = 0) {
  const m = /^data:([^;,]+);base64,([\s\S]+)$/i.exec(dataUrl.trim());
  if (!m) return null;
  const mime = (m[1] || 'image/png').toLowerCase();
  if (!mime.startsWith('image/')) return null;
  try {
    const bin = atob(m[2]);
    const bytes = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    const ext = mime.split('/')[1]?.split('+')[0] || 'png';
    return new File([bytes], `paste-${index}.${ext}`, { type: mime });
  } catch { return null; }
}

function offerDesktopPasteImages(root, files) {
  if (!files.length) return false;
  // 派发自定义事件：bubbles 让 RichEditor 内层也能监听；cancelable 让其可 preventDefault
  const ev = new CustomEvent('rich-editor:desktop-paste-images', {
    bubbles: true, cancelable: true, detail: { files },
  });
  root.dispatchEvent(ev);
  return ev.defaultPrevented; // 被拦截则 Host 不再走 base64 回退
}
