### 12.2.3 分屏跟随滚动

```typescript
// components/design/Monaco/utils.ts
interface MarkdownScrollSyncSnapshot {
  editorY: number[];     // 编辑器标题行 Y 坐标
  previewY: number[];    // 预览标题 Y 坐标
  lineCount: number;      // 文档行数
  scrollHeight: number;  // 预览区滚动高度
}

export function buildMarkdownScrollSyncSnapshot(
  editor: editor.IStandaloneCodeEditor,
  previewContainer: HTMLElement,
): MarkdownScrollSyncSnapshot {
  // 采集标题 DOM
  const headingLines = Array.from(
    previewContainer.querySelectorAll('[data-md-heading-line]'),
  ) as HTMLElement[];

  const editorY: number[] = [];
  const previewY: number[] = [];

  for (const heading of headingLines) {
    const lineNumber = parseInt(heading.dataset.mdHeadingLine || '0');
    const top = heading.offsetTop;

    editorY.push(editor.getTopForLineNumber(lineNumber));
    previewY.push(top);
  }

  // 确保单调递增
  const pairs = editorY.map((y, i) => ({ y, target: previewY[i] }))
    .sort((a, b) => a.y - b.y);

  return {
    editorY: pairs.map(p => p.y),
    previewY: pairs.map(p => p.target),
    lineCount: editor.getModel()?.getLineCount() || 0,
    scrollHeight: previewContainer.scrollHeight,
  };
}

export function syncPreviewScrollFromMarkdownEditor(
  editor: editor.IStandaloneCodeEditor,
  preview: HTMLElement,
  snapshot: MarkdownScrollSyncSnapshot,
  deadband = 1.5,
) {
  const scrollTop = editor.getScrollTop();

  // 分段线性插值
  let targetPreviewScrollTop = 0;
  const { editorY, previewY } = snapshot;

  for (let i = 0; i < editorY.length - 1; i++) {
    if (scrollTop >= editorY[i] && scrollTop <= editorY[i + 1]) {
      const ratio = (scrollTop - editorY[i]) / (editorY[i + 1] - editorY[i]);
      targetPreviewScrollTop = previewY[i] + ratio * (previewY[i + 1] - previewY[i]);
      break;
    }
  }

  // 死区抑制
  const diff = Math.abs(targetPreviewScrollTop - preview.scrollTop);
  if (diff > deadband) {
    preview.scrollTo({ top: targetPreviewScrollTop, behavior: 'auto' });
  }
}
```

### 12.2.4 Mermaid 岛布局

```typescript
// packages/markdown-kit/src/markdown/parser.ts
class MarkdownParser {
  splitForMermaidIslands(text: string): MarkdownMermaidSplitPart[] {
    const parts: MarkdownMermaidSplitPart[] = [];
    const lines = text.split('\n');
    let buffer: string[] = [];
    let inMermaidBlock = false;

    for (const line of lines) {
      if (line.trim().startsWith('```mermaid')) {
        // 保存之前的 markdown 段
        if (buffer.length) {
          parts.push({ type: 'markdown', content: buffer.join('\n') });
          buffer = [];
        }
        inMermaidBlock = true;
      } else if (inMermaidBlock && line.trim() === '```') {
        parts.push({
          type: 'mermaid',
          content: buffer.join('\n'),
        });
        buffer = [];
        inMermaidBlock = false;
      } else {
        buffer.push(line);
      }
    }

    if (buffer.length) {
      parts.push({
        type: inMermaidBlock ? 'mermaid' : 'markdown',
        content: buffer.join('\n'),
      });
    }

    return parts;
  }
}
```

## 12.3 Mermaid 渲染

### 12.3.1 两条渲染路径

```
路径 A：整段 HTML
    Markdown → parser.render → dangerouslySetInnerHTML
    → useMermaidInMarkdownRoot（双 rAF）→ querySelectorAll('.mermaid') → mermaid.run

路径 B：围栏拆分 + Mermaid 岛
    Markdown → parser.splitForMermaidIslands
    → markdown 段：parser.render(enableMermaid: false) → innerHTML
    → mermaid 段：MermaidFenceIsland → 离屏渲染 → 成功提交 SVG
```

### 12.3.2 MermaidFenceIsland

```typescript
// components/design/MermaidFenceIsland.tsx
export function MermaidFenceIsland({ code, onError }: Props) {
  const [svg, setSvg] = useState<string | null>(null);
  const [status, setStatus] = useState<'loading' | 'rendering' | 'ready' | 'error'>('loading');

  useEffect(() => {
    let cancelled = false;

    async function render() {
      setStatus('rendering');
      try {
        // 离屏渲染：避免错误 SVG 闪烁
        const container = document.createElement('div');
        container.style.position = 'absolute';
        container.style.visibility = 'hidden';

        // mermaid.run 全局队列
        const { svg } = await mermaid.render(`mermaid-${Date.now()}`, code);

        if (!cancelled) {
          // 成功才提交 SVG
          setSvg(svg);
          setStatus('ready');
        }
      } catch (e) {
        if (!cancelled) {
          setStatus('error');
          onError?.(e);
        }
      }
    }

    render();
    return () => { cancelled = true; };
  }, [code]);

  if (status === 'error') {
    return <div className="text-destructive text-sm">Mermaid 渲染失败</div>;
  }

  return (
    <div
      className="mermaid-island overflow-auto"
      dangerouslySetInnerHTML={svg ? { __html: svg } : undefined}
    />
  );
}
```

// hooks/useMermaidDiagramClickPreview.tsx
export function useMermaidDiagramClickPreview(rootRef: RefObject<HTMLElement>) {
  const handleClick = useCallback((e: MouseEvent) => {
    const target = e.target as HTMLElement;
    const svg = target.closest('svg.mermaid');
    if (!svg) return;

    // 获取 SVG 内容
    const serializer = new XMLSerializer();
    const svgString = serializer.serializeToString(svg);
    const dataUrl = `data:image/svg+xml;charset=utf-8,${encodeURIComponent(svgString)}`;

    // 打开预览
    showImagePreview(dataUrl);
  }, []);

  useEffect(() => {
    const root = rootRef.current;
    if (!root) return;

    root.addEventListener('click', handleClick);
    return () => root.removeEventListener('click', handleClick);
  }, [rootRef, handleClick]);
}

// components/RichEditor/extensions/TitleNode.ts
export const TitleNode = Node.create({
  name: 'title',
  group: 'block',
  atom: true,
  content: 'title block+',
  selectable: true,
  draggable: false,

  addAttributes() {
    return {
      level: { default: 1 },
      text: { default: '' },
    };
  },

  addCommands() {
    return {
      setTitle: (text: string, level?: number) => ({ commands }) => {
        return commands.insertContent({
          type: 'title',
          attrs: { text, level: level || 1 },
        });
      },
    };
  },

  addTransaction() {
    return (tr) => {
      // 监听删多余 title
      tr.docChanged = (tr.docChanged as any).concat((transaction) => {
        const titleCount = transaction.doc.descendants('node => node.type.name === 'title')
          .length;
        if (titleCount > 1) {
          // 只保留第一个 title
          const titles = transaction.doc.descendants(
            (node, pos) => node.type.name === 'title' ? { node, pos } : false,
          );
          for (const { pos } of titles.slice(1)) {
            transaction.delete(pos, pos + 1);
          }
        }
        return transaction;
      });
    };
  },
});

// components/RichEditor/image/ImageUpload.ts
export const ImageUpload = Extension.create({
  name: 'imageUpload',

  addOptions() {
    return {
      uploader: null as ((file: File) => Promise<string>) | null,
    };
  },

  addProps() {
    return {
      handlePaste: (view: EditorView, event: ClipboardEvent) => {
        const files = Array.from(event.clipboardData?.items || []);
        const imageFiles = files.filter(f => f.type.startsWith('image/'));

        for (const file of imageFiles) {
          this.handleFile(file);
        }
      },
      handleDrop: (view: EditorView, event: DragEvent) => {
        const files = Array.from(event.dataTransfer?.files || []);
        for (const file of files) {
          this.handleFile(file);
        }
      },
    };
  },

  addMethods() {
    return {
      async handleFile(file: File) {
        const url = await this.options.uploader?.(file);
        if (url) {
          this.editor.chain().focus().setImage({ src: url }).run();
        }
      },
    };
  },
});

// components/RichEditor/extensions/CharacterCount.ts
import { Intl } from 'intl';

export const CharacterCount = Extension.create({
  name: 'characterCount',

  addStates() {
    return {
      count: 0,
    };
  },

  addMethods() {
    return {
      getCount() {
        const text = this.editor.state.doc.textContent;
        // grapheme 粒度计数
        const segmenter = new Intl.Segmenter(undefined, { granularity: 'grapheme' });
        const segments = segmenter.segment(text);
        return segments.length;
      },
    };
  },

  addKeyboardShortcuts() {
    return {
      'Mod-s': () => {
        this.editor.storage.characterCount.updateCount();
        return true;
      },
    };
  },
});
