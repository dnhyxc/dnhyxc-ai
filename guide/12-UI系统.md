### 12.2.3 分屏跟随滚动

```typescript
// components/design/Monaco/utils.ts
interface MarkdownScrollSyncSnapshot {
  editorY: number[];     // 编辑器标题行 Y 坐标
  previewY: number[];    // 预览标题 Y 坐标
  lineCount: number;      // 文档行数
  scrollHeight: number;  // 预览区滚动高度
}

export function buildMarkdownScrollSyncSnapshot(
  editor: editor.IStandaloneCodeEditor,
  previewContainer: HTMLElement,
): MarkdownScrollSyncSnapshot {
  // 采集标题 DOM
  const headingLines = Array.from(
    previewContainer.querySelectorAll('[data-md-heading-line]'),
  ) as HTMLElement[];

  const editorY: number[] = [];
  const previewY: number[] = [];

  for (const heading of headingLines) {
    const lineNumber = parseInt(heading.dataset.mdHeadingLine || '0');
    const top = heading.offsetTop;

    editorY.push(editor.getTopForLineNumber(lineNumber));
    previewY.push(top);
  }

  // 确保单调递增
  const pairs = editorY.map((y, i) => ({ y, target: previewY[i] }))
    .sort((a, b) => a.y - b.y);

  return {
    editorY: pairs.map(p => p.y),
    previewY: pairs.map(p => p.target),
    lineCount: editor.getModel()?.getLineCount() || 0,
    scrollHeight: previewContainer.scrollHeight,
  };
}

export function syncPreviewScrollFromMarkdownEditor(
  editor: editor.IStandaloneCodeEditor,
  preview: HTMLElement,
  snapshot: MarkdownScrollSyncSnapshot,
  deadband = 1.5,
) {
  const scrollTop = editor.getScrollTop();

  // 分段线性插值
  let targetPreviewScrollTop = 0;
  const { editorY, previewY } = snapshot;

  for (let i = 0; i < editorY.length - 1; i++) {
    if (scrollTop >= editorY[i] && scrollTop <= editorY[i + 1]) {
      const ratio = (scrollTop - editorY[i]) / (editorY[i + 1] - editorY[i]);
      targetPreviewScrollTop = previewY[i] + ratio * (previewY[i + 1] - previewY[i]);
      break;
    }
  }

  // 死区抑制
  const diff = Math.abs(targetPreviewScrollTop - preview.scrollTop);
  if (diff > deadband) {
    preview.scrollTo({ top: targetPreviewScrollTop, behavior: 'auto' });
  }
}
```

### 12.2.4 Mermaid 岛布局

```typescript
// packages/markdown-kit/src/markdown/parser.ts
class MarkdownParser {
  splitForMermaidIslands(text: string): MarkdownMermaidSplitPart[] {
    const parts: MarkdownMermaidSplitPart[] = [];
    const lines = text.split('\n');
    let buffer: string[] = [];
    let inMermaidBlock = false;

    for (const line of lines) {
      if (line.trim().startsWith('```mermaid')) {
        // 保存之前的 markdown 段
        if (buffer.length) {
          parts.push({ type: 'markdown', content: buffer.join('\n') });
          buffer = [];
        }
        inMermaidBlock = true;
      } else if (inMermaidBlock && line.trim() === '```') {
        parts.push({
          type: 'mermaid',
          content: buffer.join('\n'),
        });
        buffer = [];
        inMermaidBlock = false;
      } else {
        buffer.push(line);
      }
    }

    if (buffer.length) {
      parts.push({
        type: inMermaidBlock ? 'mermaid' : 'markdown',
        content: buffer.join('\n'),
      });
    }

    return parts;
  }
}

## 12.3 Mermaid 渲染

### 12.3.1 两条渲染路径

```
路径 A：整段 HTML
    Markdown → parser.render → dangerouslySetInnerHTML
    → useMermaidInMarkdownRoot（双 rAF）→ querySelectorAll('.mermaid') → mermaid.run

路径 B：围栏拆分 + Mermaid 岛
    Markdown → parser.splitForMermaidIslands
    → markdown 段：parser.render(enableMermaid: false) → innerHTML
    → mermaid 段：MermaidFenceIsland → 离屏渲染 → 成功提交 SVG
```

### 12.3.2 MermaidFenceIsland

```typescript
// components/design/MermaidFenceIsland.tsx
export function MermaidFenceIsland({ code, onError }: Props) {
  const [svg, setSvg] = useState<string | null>(null);
  const [status, setStatus] = useState<'loading' | 'rendering' | 'ready' | 'error'>('loading');

  useEffect(() => {
    let cancelled = false;

    async function render() {
      setStatus('rendering');
      try {
        // 离屏渲染：避免错误 SVG 闪烁
        const container = document.createElement('div');
        container.style.position = 'absolute';
        container.style.visibility = 'hidden';

        // mermaid.run 全局队列
        const { svg } = await mermaid.render(`mermaid-${Date.now()}`, code);

        if (!cancelled) {
          // 成功才提交 SVG
          setSvg(svg);
          setStatus('ready');
        }
      } catch (e) {
        if (!cancelled) {
          setStatus('error');
          onError?.(e);
        }
      }
    }

    render();
    return () => { cancelled = true; };
  }, [code]);

  if (status === 'error') {
    return <div className="text-destructive text-sm">Mermaid 渲染失败</div>;
  }

  return (
    <div
      className="mermaid-island overflow-auto"
      dangerouslySetInnerHTML={svg ? { __html: svg } : undefined}
    />
  );
}
```

// hooks/useMermaidDiagramClickPreview.tsx
export function useMermaidDiagramClickPreview(rootRef: RefObject<HTMLElement>) {
  const handleClick = useCallback((e: MouseEvent) => {
    const target = e.target as HTMLElement;
    const svg = target.closest('svg.mermaid');
    if (!svg) return;

    // 获取 SVG 内容
    const serializer = new XMLSerializer();
    const svgString = serializer.serializeToString(svg);
    const dataUrl = `data:image/svg+xml;charset=utf-8,${encodeURIComponent(svgString)}`;

    // 打开预览
    showImagePreview(dataUrl);
  }, []);

  useEffect(() => {
    const root = rootRef.current;
    if (!root) return;

    root.addEventListener('click', handleClick);
    return () => root.removeEventListener('click', handleClick);
  }, [rootRef, handleClick]);
}

// components/RichEditor/extensions/TitleNode.ts
export const TitleNode = Node.create({
  name: 'title',
  group: 'block',
  atom: true,
  content: 'title block+',
  selectable: true,
  draggable: false,

  addAttributes() {
    return {
      level: { default: 1 },
      text: { default: '' },
    };
  },

  addCommands() {
    return {
      setTitle: (text: string, level?: number) => ({ commands }) => {
        return commands.insertContent({
          type: 'title',
          attrs: { text, level: level || 1 },
        });
      },
    };
  },

  addTransaction() {
    return (tr) => {
      // 监听删多余 title
      tr.docChanged = (tr.docChanged as any).concat((transaction) => {
        const titleCount = transaction.doc.descendants('node => node.type.name === 'title')
          .length;
        if (titleCount > 1) {
          // 只保留第一个 title
          const titles = transaction.doc.descendants(
            (node, pos) => node.type.name === 'title' ? { node, pos } : false,
          );
          for (const { pos } of titles.slice(1)) {
            transaction.delete(pos, pos + 1);
          }
        }
        return transaction;
      });
    };
  },
});

## 12.9 Sheet 抽屉描边与浅色主题墨色基线

### 12.9.1 两个常见视觉问题

接入 shadcn/ui 的 `Sheet` 组件后，浅色主题下常出现两个现象：
1. **四个方向的抽屉（尤其 right）没有锚点感**，看起来像浮在白底上的纯白块——原实现只在 right 侧写了 `border-l border-theme-border`（近白灰），其它三面无线；
2. **所有使用 `border-theme/N`、`bg-theme/N` 的卡片/弹窗在浅色下完全透明**——原因是 `:root` 默认浅色主题错把 `--theme-color` 写成了近白值 `oklch(100% ...)`，墨色 + 10% 透明叠在白底上对比度为 0。

### 12.9.2 token 语义：`--theme-color` 是「墨色」不是「背景色」

全站 UI 语义：
- `--theme-color`：通用墨色基线，Tailwind 的 `border-theme/N`、`bg-theme/N`、`text-theme/N` 都依赖它做半透明叠加；浅色下应为深色、暗色下应为近白；
- `--theme-background` / `--theme-card`：纯白或浅色底；
- `--theme-white`：永远纯白（显式取白场景专用）。

**修复**：将 `:root` 浅色默认下的 `--theme-color` 与 `--theme-textcolor` 纠正为墨色 `oklch(0.15 0.02 264.665)`（与 `.theme-white` 语义一致），并在上方加防守性注释说明「不能是白」。注意：`--theme-text-from` / `--theme-text-to`（Hero 渐变文字用）保持白不变，它们是渐变起止色，场景与正文文本不同。

```css
/* 与 .theme-white 对齐：theme-color 是描边/淡底用的「墨色」，不能是白
   否则 border-theme/10 在浅底上全透明、bg-theme/N 也看不出底色 */
--theme-color: oklch(0.15 0.02 264.665);
/* 正文色：与墨色基线对齐，浅色下可读 */
--theme-textcolor: oklch(0.15 0.02 264.665);
/* theme-white 常量：永远纯白（显式取白用，避免随 theme-color 漂移） */
--theme-white: oklch(1 0 0);
```

### 12.9.3 Sheet 父容器统一补描边 + 分支只声明宽度

旧版 `SheetContent` 的公共 `cn()` className 不带边框，只有 `side === 'right'` 分支写了 `border-l border-theme-border`。改为：

1. **公共 className 新增 `border-theme/10`**，让四个方向统一带 10% 墨色描边；
2. 各侧分支只保留 `border-l / border-r / border-t / border-b`（声明「哪一侧画线」的宽度，不写颜色）；
3. 移除 `side === 'right'` 专属的 `border-theme-border`，颜色由父级 token 一致供给。

这样四侧边界统一、切换主题/强调色时描边自然联动，不会出现「right 侧有硬白边、其他三面无」的不一致观感。

```tsx
<SheetPrimitive.Content
  data-slot="sheet-content"
  className={cn(
    // 基础：四面统一 10% 墨色描边（修复浅色下无锚点）
    '[&>button]:hidden bg-theme-background border-theme/10 data-[state=open]:animate-in data-[state=closed]:animate-out fixed z-50 flex flex-col gap-4 shadow-lg transition ease-in-out data-[state=closed]:duration-300 data-[state=open]:duration-500',
    // right 侧：只写 border-l（宽度 1px），颜色继承父级 border-theme/10
    side === 'right' &&
      'data-[state=closed]:slide-out-to-right data-[state=open]:slide-in-from-right inset-y-0 right-0 h-full w-3/4 border-l sm:max-w-sm',
    // left 侧：border-r（宽度），颜色同样继承父级
    side === 'left' &&
      'data-[state=closed]:slide-out-to-left data-[state=open]:slide-in-from-left inset-y-0 left-0 h-full w-3/4 border-r sm:max-w-sm',
    // top 侧：border-b（宽度）
    side === 'top' &&
      'data-[state=closed]:slide-out-to-top data-[state=open]:slide-in-from-top inset-x-0 top-0 h-auto border-b',
    // bottom 侧：border-t（宽度）
    side === 'bottom' &&
      'data-[state=closed]:slide-out-to-bottom data-[state=open]:slide-in-from-bottom inset-x-0 bottom-0 h-auto border-t',
    // 用户自定义 className（最高优先级覆盖）
    className,
  )}
  {...props}
>
```

### 12.9.4 行为与兼容性

| 区域 | 旧表现 | 新表现 |
|---|---|---|
| Sheet 浅色 | right 侧有近白灰线、其它 3 侧无 | 四面统一柔和墨色描边 |
| `border-theme/10` 全局 | 浅色下 ≈ 全透明（白叠白） | 真实描边线，卡片/弹窗不浮空 |
| `bg-theme/[5–20]` 全局 | 无底色反馈 | 淡淡墨底，hover/激活态真实 |
| `text-textcolor` 正文 | 曾可能白字白底 | 浅色下为深色墨，始终可读 |

- **暗色主题** `.dark` token 不在本节涉及，体验不变；
- 仍需「实线条硬边框」的场景（表单 inputs、对话框边界）继续使用 `border-theme-border`，不受本次修正影响；
- 若某卡片历史上**错误地**依赖 `--theme-color = 白`（如裸写 `bg-theme` 当背景白），应改为语义正确的 `bg-theme-background` 或 `bg-theme-white`。

### 12.9.5 回归建议

- **Sheet 四方向**：分别打开设置抽屉（right）、电子书目录、移动端顶部提示（top）、底部操作面板（bottom），浅色下确认四面都有柔和描边；
- **卡片不再浮空**：浏览插件中心、知识库列表、电子书书架等 Grid 页面，浅色下确认每张卡片带淡边缘；
- **正文可读**：首页欢迎语、设置页说明、插件描述确认文字非白；
- **渐变字不被误改**：首页 Hero 渐变标题仍为白色渐变，未变墨色；
- **暗色无多余亮边**：切暗色主题，确认 Sheet 与卡片不会多出异常亮边。
