# 十三、国际化与设置

## 13.1 国际化（i18n）

### 13.1.1 useSyncExternalStore 驱动

```typescript
// hooks/i18n.ts
import { useSyncExternalStore } from 'react';

// 模块级状态
let currentLocale: Locale = getInitialLocale();
const localeListeners = new Set<() => void>();

function emitLocaleChanged() {
  localeListeners.forEach(fn => fn());
}

function subscribe(listener: () => void) {
  localeListeners.add(listener);
  return () => {
    localeListeners.delete(listener);
  };
}

function getSnapshot() {
  return currentLocale;
}

export function useI18n() {
  const locale = useSyncExternalStore(subscribe, getSnapshot, getSnapshot);

  const setLocale = (newLocale: Locale, opts?: { syncUrl?: boolean }) => {
    currentLocale = newLocale;

    // 持久化
    localStorage.setItem('dnhyxc_locale_bootstrap', newLocale);

    // 同步 URL
    if (opts?.syncUrl) {
      const url = new URL(window.location.href);
      url.searchParams.set('lang', newLocale);
      window.history.replaceState({}, '', url.toString());
    }

    emitLocaleChanged();
  };

  const t = useCallback((key: string, params?: Record<string, string>) => {
    const dict = DICTS[locale] || DICTS[DEFAULT_LOCALE];
    let template = dict[key] || DICTS[DEFAULT_LOCALE][key] || key;

    // 参数插值
    if (params) {
      template = template.replace(
        /\{(\w+)\}/g,
        (_, k) => params[k] ?? `{${k}}`,
      );
    }

    return template;
  }, [locale]);

  return {
    locale,
    t,
    setLocale,
    toggleLocale: () => {
      setLocale(locale === 'zh-CN' ? 'en-US' : 'zh-CN', { syncUrl: true });
    },
    supportedLocales: SUPPORTED_LOCALES,
  };
}
```

### 13.1.2 初始化优先级

```typescript
function getInitialLocale(): Locale {
  // 1. URL 参数
  const urlParams = new URLSearchParams(window.location.search);
  const langFromUrl = urlParams.get('lang') as Locale | null;
  if (langFromUrl && SUPPORTED_LOCALES.includes(langFromUrl)) {
    return langFromUrl;
  }

  // 2. localStorage bootstrap
  const bootstrap = localStorage.getItem('dnhyxc_locale_bootstrap') as Locale | null;
  if (bootstrap && SUPPORTED_LOCALES.includes(bootstrap)) {
    return bootstrap;
  }

  // 3. Tauri 异步持久化（通过 invoke 获取）
  // 注意：不能在模块加载时调用 invoke，需在 useEffect 中
  // 此处仅作降级处理

  // 4. 默认
  return DEFAULT_LOCALE;
}
```

```typescript
function applyAccentToDocument(colorId: string) {
  const color = ACCENT_COLORS.find(c => c.id === colorId);
  if (!color) return;

  const root = document.documentElement;

  // 主色
  root.style.setProperty('--brand-accent', color.hex);

  // 衍生色（使用 color-mix 现场计算）
  root.style.setProperty('--brand-accent-soft',
    colorMix('oklch', color.hex, 'white', 0.85));
  root.style.setProperty('--brand-accent-light',
    colorMix('oklch', color.hex, 'white', 0.6));
  root.style.setProperty('--brand-accent-dark',
    colorMix('oklch', color.hex, 'black', 0.2));

  // 覆盖 Tailwind teal 类
  // text-teal-500 → text-[var(--brand-accent)]
  // bg-teal-500 → bg-[var(--brand-accent)]
  // bg-teal-500/15 → bg-[var(--brand-accent)]/15
}
```

```typescript
function persistAccentBootstrap(colorId: string) {
  const color = ACCENT_COLORS.find(c => c.id === colorId);
  if (!color) return;

  // localStorage（首帧读取）
  localStorage.setItem('dnhyxc_accent_bootstrap', JSON.stringify({
    id: color.id,
    hex: color.hex,
  }));

  // Tauri store（跨端同步）
  if (isTauriRuntime()) {
    invoke('set_store', {
      key: 'accentColor',
      value: colorId,
    });
  }
}
```

### 13.3.2 Rust 侧解析

```rust
// src-tauri/src/system/shortcut.rs
pub fn parse_chord(chord: &str) -> Result<(Vec<KeyCode>, Modifiers), String> {
    let parts: Vec<&str> = chord.split('+').collect();
    let mut keys = Vec::new();
    let mut modifiers = Modifiers::empty();

    for part in parts {
        match part.to_lowercase().as_str() {
            "meta" | "cmd" | "command" => modifiers |= Modifiers::SUPER,
            "ctrl" | "control" => modifiers |= Modifiers::CONTROL,
            "shift" => modifiers |= Modifiers::SHIFT,
            "alt" | "option" => modifiers |= Modifiers::ALT,
            _ => {
                if let Ok(key) = parse_key_code(part) {
                    keys.push(key);
                }
            }
        }
    }

    Ok((keys, modifiers))
}
```

### 13.3.3 冲突检测

```typescript
// views/setting/system/index.tsx
function chordStringsSemanticallyEqual(a: string, b: string): boolean {
  const normalize = (s: string) => s
    .toLowerCase()
    .split('+')
    .map(k => k.trim())
    .sort()
    .join('+');

  return normalize(a) === normalize(b);
}

function detectConflicts(newShortcut: string, currentKey: string): string | null {
  for (const item of DEFAULT_INFO) {
    if (item.key !== currentKey && chordStringsSemanticallyEqual(item.shortcut, newShortcut)) {
      return item.labelKey;
    }
  }
  return null;
}
```
