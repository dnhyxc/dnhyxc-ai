### 15.3.2 向量化流程

```typescript
// services/embedding.service.ts
@Injectable()
export class EmbeddingService {
  async embedChunks(chunks: string[]): Promise<number[][]> {
    const response = await axios.post(
      `${this.config.get('llm.providers.volcengine.baseUrl')}/embeddings`,
      {
        model: 'text-embedding-3-small',
        input: chunks,
      },
      {
        headers: {
          Authorization: `Bearer ${this.config.get('llm.providers.volcengine.apiKey')}`,
        },
      },
    );

    return response.data.data.map((d: any) => d.embedding);
  }

  async indexDocument(documentId: string) {
    const chunks = await this.chunkRepo.find({ where: { documentId } });

    // 分批向量化（100 条/批）
    const batchSize = 100;
    for (let i = 0; i < chunks.length; i += batchSize) {
      const batch = chunks.slice(i, i + batchSize);
      const texts = batch.map(c => c.content);
      const vectors = await this.embedChunks(texts);

      // 写入 Qdrant
      const points = batch.map((chunk, j) => ({
        id: chunk.id,
        vector: vectors[j],
        payload: {
          document_id: documentId,
          chunk_index: i + j,
          user_id: chunk.userId,
        },
      }));

      await this.qdrant.upsert('knowledge_chunks', points);
    }

    // 更新索引状态
    document.isIndexed = true;
    document.indexedAt = new Date();
    await this.docRepo.save(document);
  }
}
```

### 15.3.3 混合检索

```typescript
// services/retrieval.service.ts
@Injectable()
export class RetrievalService {
  async hybridSearch(
    query: string,
    options: {
      userId: string;
      topK?: number;
      vectorWeight?: number;
      bm25Weight?: number;
    },
  ): Promise<SearchResult[]> {
    const topK = options.topK || 10;
    const vectorWeight = options.vectorWeight ?? 0.7;
    const bm25Weight = options.bm25Weight ?? 0.3;

    // 1. 向量检索
    const queryVector = await this.embedding.embedChunks([query]);
    const vectorResults = await this.qdrant.search('knowledge_chunks', {
      vector: queryVector[0],
      limit: topK * 2,
      filter: { must: [{ key: 'user_id', match: { value: options.userId } }] },
    });

    // 2. BM25 关键词检索
    const bm25Results = await this.bm25Search(query, options.userId, topK * 2);

    // 3. 分数融合（RRF）
    const fused = this.rrfFusion(vectorResults, bm25Results, vectorWeight, bm25Weight);

    // 4. 去重 + 排序
    const deduplicated = this.deduplicateByDocument(fused, topK);

    return deduplicated;
  }

  private rrfFusion(
    vectorResults: QdrantPoint[],
    bm25Results: Bm25Hit[],
    vectorWeight: number,
    bm25Weight: number,
  ): FusionResult[] {
    const scores = new Map<string, number>();

    // 向量分数归一化
    const maxVectorScore = Math.max(...vectorResults.map(r => r.score));
    for (let i = 0; i < vectorResults.length; i++) {
      const id = vectorResults[i].id;
      const normalized = (vectorResults[i].score / maxVectorScore) * vectorWeight;
      scores.set(id, (scores.get(id) || 0) + normalized);
    }

    // BM25 分数归一化
    const maxBm25Score = Math.max(...bm25Results.map(r => r.score));
    for (let i = 0; i < bm25Results.length; i++) {
      const id = bm25Results[i].id;
      const normalized = (bm25Results[i].score / maxBm25Score) * bm25Weight;
      scores.set(id, (scores.get(id) || 0) + normalized);
    }

    return Array.from(scores.entries())
      .map(([id, score]) => ({ id, score }))
      .sort((a, b) => b.score - a.score);
  }
}
```

### 15.4.2 同步队列

```typescript
// offline/sync-queue.ts
class SyncQueue {
  async enqueue(operation: SyncOperation): Promise<void> {
    const db = await openDB(DB_NAME, DB_VERSION);
    await db.add('sync_queue', {
      ...operation,
      status: 'pending',
      createdAt: Date.now(),
      retries: 0,
    });
  }

  async processQueue(): Promise<void> {
    const db = await openDB(DB_NAME, DB_VERSION);
    const pending = await db.getAllFromIndex('sync_queue', 'status', 'pending');

    for (const item of pending) {
      try {
        await this.executeOperation(item);
        await db.delete('sync_queue', item.id);
      } catch (error) {
        await db.update('sync_queue', item.id, {
          status: item.retries >= 3 ? 'failed' : 'pending',
          retries: item.retries + 1,
          lastError: error.message,
        });
      }
    }
  }

  private async executeOperation(op: SyncOperation): Promise<void> {
    switch (op.type) {
      case 'create_message':
        return api.post('/chat/messages', op.payload);
      case 'update_document':
        return api.put(`/knowledge/documents/${op.id}`, op.payload);
      case 'delete_document':
        return api.delete(`/knowledge/documents/${op.id}`);
      default:
        throw new Error(`Unknown operation: ${op.type}`);
    }
  }
}
```

## 15.5 Redis 缓存

### 15.5.1 缓存策略

```typescript
// services/cache.service.ts
@Injectable()
export class CacheService {
  async getOrSet<T>(
    key: string,
    fetchFn: () => Promise<T>,
    ttlSeconds: number = 3600,
  ): Promise<T> {
    const cached = await this.redis.get(key);
    if (cached) {
      return JSON.parse(cached) as T;
    }

    const value = await fetchFn();
    await this.redis.setex(key, ttlSeconds, JSON.stringify(value));
    return value;
  }

  async invalidateByPattern(pattern: string): Promise<void> {
    const keys = await this.redis.keys(pattern);
    if (keys.length > 0) {
      await this.redis.del(keys);
    }
  }
}
```

### 15.5.2 限流

```typescript
// services/rate-limit.service.ts
@Injectable()
export class RateLimitService {
  async checkRateLimit(
    key: string,
    limit: number,
    windowSeconds: number,
  ): Promise<boolean> {
    const current = await this.redis.incr(`ratelimit:${key}`);
    if (current === 1) {
      await this.redis.expire(`ratelimit:${key}`, windowSeconds);
    }
    return current <= limit;
  }
}
```

```typescript
@Cron('0 2 * * *')  // 每天凌晨 2 点
export class BackupJob {
  async run() {
    // 1. 导出 PostgreSQL
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const backupPath = `/backups/${timestamp}.sql`;
    await exec(`pg_dump -h ${DB_HOST} -U ${DB_USER} ${DB_NAME} > ${backupPath}`);

    // 2. 压缩上传
```
